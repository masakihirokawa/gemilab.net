## Function Calling とは何か — AIをリアルな世界につなぐ仕組み

Gemini API の **Function Calling**（ツール呼び出し）は、AIモデルが外部の関数やAPIと連携できるようにする強力な機能です。通常、LLMはテキストを生成するだけですが、Function Calling を使うと「天気を調べる」「データベースを検索する」「カレンダーに予定を追加する」といった実際のアクションをトリガーできるようになります。

2026年現在、Gemini 2.5 Pro・Gemini 2.5 Flash・Gemini 3 シリーズすべてで Function Calling がサポートされており、個人開発者からエンタープライズまで幅広く活用されています。

Function Calling の動作フローを簡単に説明すると次の通りです。

1. 開発者が「使える関数」の定義をAPIに渡す
2. ユーザーが「東京の天気は？」と入力する
3. Gemini が「この質問には `get_weather(city="Tokyo")` を呼ぶべき」と判断し、関数呼び出し情報を返す
4. アプリ側で実際に関数を実行し、結果をモデルに渡す
5. Gemini が結果をもとに自然な回答を生成する

この「AIが判断し、アプリが実行する」という分業こそが Function Calling の本質です。

## 最初のFunction Calling実装 — 天気取得の例

Function Calling の基本を学ぶために、天気取得のサンプルを実装してみましょう。実際の天気APIの代わりに、ダミーデータを返す関数を使います。

### Step 1: 関数の定義（ツール定義）

```python
import json
from google import genai
from google.genai import types

# 実際に呼び出される Python 関数（ダミー実装）
def get_weather(city: str) -> dict:
    """指定した都市の天気情報を返す（ダミーデータ）"""
    weather_data = {
        "Tokyo": {"condition": "晴れ", "temperature": 22, "humidity": 55},
        "Osaka": {"condition": "曇り", "temperature": 19, "humidity": 68},
        "Sapporo": {"condition": "雪", "temperature": -3, "humidity": 85},
    }
    return weather_data.get(city, {"condition": "不明", "temperature": 0, "humidity": 0})

# Gemini に渡すツール定義
weather_tool = types.Tool(
    function_declarations=[
        types.FunctionDeclaration(
            name="get_weather",
            description="指定した都市の現在の天気情報（天気の状態・気温・湿度）を取得する",
            parameters=types.Schema(
                type=types.Type.OBJECT,
                properties={
                    "city": types.Schema(
                        type=types.Type.STRING,
                        description="天気を調べたい都市名（例: Tokyo, Osaka）"
                    )
                },
                required=["city"]
            )
        )
    ]
)
```

### Step 2: モデルへのリクエストと関数呼び出しの受け取り

```python
client = genai.Client(api_key=os.environ["YOUR_GEMINI_API_KEY"])

# ユーザーの質問を送信（ツール定義を同時に渡す）
response = client.models.generate_content(
    model="gemini-2.5-flash",
    contents="東京の今日の天気を教えてください",
    config=types.GenerateContentConfig(
        tools=[weather_tool]
    )
)

# Gemini がツール呼び出しを返したか確認
for part in response.candidates[0].content.parts:
    if part.function_call:
        func_call = part.function_call
        print(f"関数名: {func_call.name}")
        print(f"引数: {dict(func_call.args)}")
        # 出力例:
        # 関数名: get_weather
        # 引数: {'city': 'Tokyo'}
```

### Step 3: 関数を実行してモデルに結果を返す

```python
# 関数呼び出し情報を取得
func_call = response.candidates[0].content.parts[0].function_call

# 実際の関数を呼び出す
if func_call.name == "get_weather":
    result = get_weather(**dict(func_call.args))
    print(f"関数の実行結果: {result}")
    # 出力例: {'condition': '晴れ', 'temperature': 22, 'humidity': 55}

# 関数の結果をモデルに渡して最終回答を生成
final_response = client.models.generate_content(
    model="gemini-2.5-flash",
    contents=[
        types.Content(role="user", parts=[
            types.Part(text="東京の今日の天気を教えてください")
        ]),
        response.candidates[0].content,  # モデルの関数呼び出し
        types.Content(role="tool", parts=[
            types.Part(
                function_response=types.FunctionResponse(
                    name="get_weather",
                    response=result
                )
            )
        ])
    ],
    config=types.GenerateContentConfig(tools=[weather_tool])
)

print(final_response.text)
# 出力例: 東京の今日の天気は晴れで、気温は22°C、湿度は55%です。過ごしやすい一日になりそうですね。
```

このフローを理解することが Function Calling の第一歩です。

## 複数の関数を同時に定義する

## parallel_function_calls — 複数の関数を同時に呼ぶ

Gemini 2.5 シリーズでは、複数の関数呼び出しを**並列**に処理できます。例えば「東京と大阪の天気を同時に調べる」ような場合です。

```python
response = client.models.generate_content(
    model="gemini-2.5-flash",
    contents="東京と大阪と札幌の天気を一度に教えてください",
    config=types.GenerateContentConfig(
        tools=[weather_tool],
        # 並列呼び出しを許可（デフォルトで有効）
        automatic_function_calling=types.AutomaticFunctionCallingConfig(
            disable=False
        )
    )
)

# 複数の function_call が返される場合がある
for part in response.candidates[0].content.parts:
    if part.function_call:
        print(f"{part.function_call.name}({dict(part.function_call.args)})")
# 出力例:
# get_weather({'city': 'Tokyo'})
# get_weather({'city': 'Osaka'})
# get_weather({'city': 'Sapporo'})
```

並列呼び出しを使えば、複数のAPIリクエストを同時に処理でき、レスポンス時間を大幅に短縮できます。

## Automatic Function Calling — 自動実行モード

Google AI Python SDK では、`automatic_function_calling` を使って、関数の実行からモデルへの結果フィードバックまでを**自動化**できます。

```python
# Python 関数をそのまま渡すだけで自動的に Function Calling が動作する
def get_stock_price(ticker: str) -> float:
    """株式ティッカーシンボルの現在価格を返す（ダミーデータ）"""
    prices = {"GOOGL": 175.30, "MSFT": 420.50, "AAPL": 195.80}
    return prices.get(ticker, 0.0)

def get_company_info(ticker: str) -> dict:
    """企業情報を返す（ダミーデータ）"""
    companies = {
        "GOOGL": {"name": "Alphabet Inc.", "sector": "Technology"},
        "MSFT": {"name": "Microsoft Corp.", "sector": "Technology"},
    }
    return companies.get(ticker, {})

# 関数をリストとして渡すだけで OK
response = client.models.generate_content(
    model="gemini-2.5-flash",
    contents="Google（GOOGL）の株価と企業情報を教えてください",
    config=types.GenerateContentConfig(
        tools=[get_stock_price, get_company_info],  # 関数オブジェクトを直接渡す
    )
)

print(response.text)
# → 自動的に get_stock_price("GOOGL") と get_company_info("GOOGL") が呼び出され、
#    その結果を統合した回答が返される
```

この自動実行モードを使うと、コードが大幅にシンプルになります。プロトタイプ開発には特に便利です。

## よくあるエラーと対処法

### エラー1: 関数の戻り値が JSON シリアライズできない

```python
# ❌ 問題: datetime オブジェクトが JSON 化できない
def get_schedule(date: str) -> dict:
    from datetime import datetime
    return {"event": "会議", "time": datetime.now()}  # datetime は NG

# ✅ 解決: 文字列に変換する
def get_schedule(date: str) -> dict:
    from datetime import datetime
    return {"event": "会議", "time": datetime.now().isoformat()}  # ISO 形式の文字列
```

### エラー2: モデルが関数呼び出しをしない（テキスト回答になる）

```python
# 関数定義の description が不明確だと Gemini が呼び出しを判断できない
# ❌ 不明確な description
types.FunctionDeclaration(
    name="get_data",
    description="データを取得する",  # 何のデータか不明
    ...
)

# ✅ 具体的な description
types.FunctionDeclaration(
    name="get_weather",
    description="指定した都市の現在の気象情報（天気・気温・湿度・風速）をリアルタイムで取得する",
    ...
)
```

### エラー3: `finish_reason` が `STOP` でなく `MAX_TOKENS` になる

Function Calling のやり取りが長くなると、コンテキスト長を超えることがあります。この場合はモデルを `gemini-2.5-pro`（200万トークンコンテキスト）に変更するか、会話履歴を適切に管理してください。

## Function Calling の基本的な流れを整理する

実装に入る前に、Gemini の Function Calling がどう動くかを整理しておきます。

1. アプリが「使える関数の定義」と「ユーザーのメッセージ」を Gemini に送る
2. Gemini が必要と判断した場合、関数の呼び出し要求（`functionCall`）を返す
3. アプリが実際の関数を実行し、結果を Gemini に送り返す
4. Gemini が関数の結果を使って最終的な回答を生成する

このやり取りは複数回繰り返される可能性があります（multi-turn）。Gemini が1回目の関数呼び出し結果を受け取り、さらに別の関数を呼ぶ判断をすることもあります。

## 関数定義の設計: Gemini の動作精度に直結する

Function Calling で最も重要なのは、関数の定義（スキーマ）の書き方です。Gemini はこの定義を元に「いつ、どの関数を、どんな引数で呼ぶか」を判断します。

**曖昧な定義**（Gemini が判断を誤りやすい）:
```javascript
{
  name: "search",
  description: "検索します",
  parameters: {
    type: "object",
    properties: {
      query: { type: "string" }
    }
  }
}
```

**具体的な定義**（Gemini が正確に判断できる）:
```javascript
{
  name: "search_products",
  description: "商品データベースを検索します。商品名・カテゴリ・価格帯での絞り込みが可能です。ユーザーが特定の商品を探しているとき、または商品一覧を要求したときに使用してください。在庫確認には別の check_inventory 関数を使ってください。",
  parameters: {
    type: "object",
    properties: {
      query: {
        type: "string",
        description: "検索キーワード（商品名、カテゴリ等）"
      },
      max_price: {
        type: "number",
        description: "価格の上限（円）。指定しない場合は上限なし"
      },
      in_stock_only: {
        type: "boolean",
        description: "true の場合、在庫ありの商品のみを返す",
        default: false
      }
    },
    required: ["query"]
  }
}
```

特に重要なのは description の「いつ使うべきか」と「他の関数との違い」を明示することです。複数の関数が似た機能を持つ場合、境界を明確にしないと Gemini が誤った関数を呼ぶことがあります。

## TypeScript での実装例

```typescript
import { GoogleGenerativeAI, FunctionDeclarationSchemaType } from "@google/generative-ai";

const genAI = new GoogleGenerativeAI("YOUR_GEMINI_API_KEY");

// 関数定義
const tools = [
  {
    functionDeclarations: [
      {
        name: "get_weather",
        description:
          "指定した都市の現在の天気情報を取得します。温度・天気・湿度を返します。",
        parameters: {
          type: FunctionDeclarationSchemaType.OBJECT,
          properties: {
            city: {
              type: FunctionDeclarationSchemaType.STRING,
              description: "天気を調べる都市名（例: 東京、大阪）",
            },
            unit: {
              type: FunctionDeclarationSchemaType.STRING,
              description: "温度の単位: 'celsius' または 'fahrenheit'",
              enum: ["celsius", "fahrenheit"],
            },
          },
          required: ["city"],
        },
      },
    ],
  },
];

// 関数の実際の実装
const functions: Record<string, (args: Record<string, unknown>) => Promise<unknown>> = {
  get_weather: async ({ city, unit = "celsius" }) => {
    // 実際のAPI呼び出し
    const response = await fetchWeatherAPI(city as string, unit as string);
    return {
      city,
      temperature: response.temp,
      condition: response.weather,
      humidity: response.humidity,
    };
  },
};

// Function Calling ループ
async function chat(userMessage: string): Promise<string> {
  const model = genAI.getGenerativeModel({
    model: "gemini-2.5-pro",
    tools,
  });

  const chat = model.startChat({
    history: [],
  });

  let response = await chat.sendMessage(userMessage);

  // Function Calling ループ（最大10回で強制終了）
  let loopCount = 0;
  while (loopCount < 10) {
    const candidate = response.response.candidates?.[0];
    if (\!candidate) break;

    const functionCallParts = candidate.content.parts.filter(
      (part) => "functionCall" in part
    );

    if (functionCallParts.length === 0) {
      // 関数呼び出しなし → テキスト回答
      return response.response.text();
    }

    // 関数を実行して結果を返す
    const functionResponses = await Promise.all(
      functionCallParts.map(async (part) => {
        if (\!("functionCall" in part)) return null;
        const { name, args } = part.functionCall;

        if (\!functions[name]) {
          return {
            functionResponse: {
              name,
              response: { error: `関数 ${name} は存在しません` },
            },
          };
        }

        try {
          const result = await functions[name](args as Record<string, unknown>);
          return {
            functionResponse: { name, response: result },
          };
        } catch (error) {
          return {
            functionResponse: {
              name,
              response: { error: `実行エラー: ${String(error)}` },
            },
          };
        }
      })
    );

    response = await chat.sendMessage(
      functionResponses.filter(Boolean) as Parameters<typeof chat.sendMessage>[0]
    );
    loopCount++;
  }

  // ループ上限に達した場合
  return "処理が複雑すぎてタイムアウトしました。より具体的な質問をお試しください。";
}
```

ループカウンターは必須です。Function Calling のバグで最も危険なのは無限ループで、Gemini が同じ関数を繰り返し呼び出し続けるケースがあります。最大試行回数の上限がないと、API コストが爆発します。

## 並列 Function Calling の活用

Gemini 2.5 は複数の関数を同時に呼び出す「並列 Function Calling」をサポートしています。

```
ユーザー: 東京と大阪の天気を教えて

Gemini の応答（並列呼び出し）:
[
  { functionCall: { name: "get_weather", args: { city: "東京" } } },
  { functionCall: { name: "get_weather", args: { city: "大阪" } } }
]
```

上記のコードでは `Promise.all()` を使っているため、並列呼び出しが自然にサポートされています。2都市の天気取得は順番に実行するより並列の方が速くなります。

並列呼び出しが機能するには、関数の description が「独立した処理」であることを示している必要があります。依存関係がある処理（Aの結果を使ってBを実行する）には向きません。

## エラーハンドリングの設計

Function Calling でのエラーは2種類あります。対応が異なるため、分けて処理します。

**関数実行エラー**: 関数は呼び出せたが処理が失敗した場合。エラー内容を `response` に含めて Gemini に返すと、Gemini が状況を理解して適切な回答を生成します。

```typescript
try {
  const result = await functions[name](args);
  return { functionResponse: { name, response: result } };
} catch (error) {
  // エラーを Gemini に伝える（例外を投げない）
  return {
    functionResponse: {
      name,
      response: {
        error: "データの取得に失敗しました",
        reason: String(error),
        suggestion: "少し時間をおいてから再試行してください",
      },
    },
  };
}
```

**API エラー**: Gemini API 自体の呼び出しが失敗した場合。レート制限・タイムアウト等はリトライで対応します。

```typescript
async function sendMessageWithRetry(
  chat: ReturnType<typeof model.startChat>,
  message: Parameters<typeof chat.sendMessage>[0],
  maxRetries = 3
) {
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      return await chat.sendMessage(message);
    } catch (error) {
      if (attempt === maxRetries) throw error;
      
      // レート制限は指数バックオフ
      const isRateLimit = String(error).includes("429") || String(error).includes("RESOURCE_EXHAUSTED");
      const delay = isRateLimit ? 1000 * Math.pow(2, attempt) : 500 * attempt;
      
      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }
  throw new Error("最大リトライ回数に達しました");
}
```

## デバッグ: Function Calling 特有の問題

**Gemini が関数を呼ばない問題**

最もよく見るのは、関数が呼ばれるはずの質問に対して Gemini がテキストだけで回答するケースです。

原因として多いのは description の曖昧さです。「いつ使うべきか」の条件が明確でないと、Gemini は関数を使わずに推測で答えようとします。

診断方法: `tool_config` で関数の使用を強制します。

```typescript
const model = genAI.getGenerativeModel({
  model: "gemini-2.5-pro",
  tools,
  toolConfig: {
    functionCallingConfig: {
      mode: "ANY", // 必ず関数を呼ぶ
    },
  },
});
```

`ANY` モードで動作するなら、問題は description か `AUTO` モードでの判断にあります。description を改善するか、特定のユースケースでは `ANY` を使い続けることを検討してください。

**過剰呼び出し問題**

不要な場面でも関数が呼ばれる場合は、description に「使わない場合」を明記します。

```javascript
description: "ユーザーが具体的な商品を探しているときのみ使用してください。一般的な質問や会話には使わないでください。"
```

## 取り組みの背景 — Function Calling で Gemini の可能性を拡張する

Gemini は優れたテキスト生成能力を持っていますが、リアルタイムの天気情報や最新のデータベース内容など、モデルが持っていない情報には直接アクセスできません。この制限を突破するのが Function Calling（関数呼び出し）機能です。

Function Calling を使うと、Gemini が「この場面では外部ツールを呼び出すべきだ」と判断し、必要なパラメータを自動生成します。開発者はそのパラメータを使って実際のAPI呼び出しを行い、結果を Gemini に返すことで、リアルタイムの情報を含んだ応答を生成できます。

## Function Calling の仕組み

Function Calling は以下の3ステップで動作します。

1. **関数の定義**: 利用可能な関数（ツール）の名前、説明、パラメータを Gemini に伝える
2. **関数の呼び出し判断**: Gemini がユーザーの質問を分析し、適切な関数とパラメータを選択する
3. **結果の統合**: 関数の実行結果を Gemini に渡し、自然な文章で応答を生成する

```python
# Function Calling の全体フロー（概念図）
# ユーザー: 「東京の天気を教えて」
#   ↓
# Gemini: 関数 get_weather(city="Tokyo") を呼び出すべきと判断
#   ↓
# 開発者のコード: 天気APIを実際に呼び出す
#   ↓
# Gemini: 「東京の現在の天気は晴れ、気温は22℃です」と応答
```

## 基本実装: 天気情報の取得

最もシンプルな例として、天気情報を取得する Function Calling を実装します。

```python
import google.generativeai as genai

# APIキーの設定

# Step 1: 関数の定義
get_weather = genai.protos.FunctionDeclaration(
    name="get_weather",
    description="指定された都市の現在の天気情報を取得する",
    parameters=genai.protos.Schema(
        type=genai.protos.Type.OBJECT,
        properties={
            "city": genai.protos.Schema(
                type=genai.protos.Type.STRING,
                description="天気を取得したい都市名（例: Tokyo, New York）",
            ),
            "unit": genai.protos.Schema(
                type=genai.protos.Type.STRING,
                enum=["celsius", "fahrenheit"],
                description="温度の単位（デフォルト: celsius）",
            ),
        },
        required=["city"],
    ),
)

# ツール定義をモデルに渡す

    model_name="gemini-2.5-pro",

# Gemini が関数呼び出しを要求した場合

print(f"呼び出し関数: {function_call.name}")
print(f"パラメータ: {dict(function_call.args)}")

# 期待される出力:
# 呼び出し関数: get_weather
# パラメータ: {'city': 'Tokyo', 'unit': 'celsius'}
```

```python
import requests

def call_weather_api(city: str, unit: str = "celsius") -> dict:
    """実際の天気APIを呼び出す関数"""
    # OpenWeatherMap API の例
    api_key = "YOUR_WEATHER_API_KEY"
    url = f"https://api.openweathermap.org/data/2.5/weather"
    params = {
        "q": city,
        "appid": api_key,
        "units": "metric" if unit == "celsius" else "imperial",

    resp = requests.get(url, params=params)
    data = resp.json()
    return {
        "city": city,
        "temperature": data["main"]["temp"],
        "condition": data["weather"][0]["description"],
        "humidity": data["main"]["humidity"],
    }

# Step 3: 関数を実行して結果を Gemini に返す
```

## エラーハンドリングとベストプラクティス

### 関数呼び出しのバリデーション

```python
def handle_function_call(response) -> dict | None:
    """Function Call のレスポンスを安全に処理する"""
    try:
        candidate = response.candidates[0]
        part = candidate.content.parts[0]

        # Function Call が含まれているか確認
        if not hasattr(part, "function_call") or not part.function_call.name:
            # 通常のテキスト応答の場合
            return None

        fc = part.function_call
        return {
            "name": fc.name,
            "args": dict(fc.args),
        }
    except (IndexError, AttributeError) as e:
        print(f"Function Call の解析に失敗: {e}")
        return None

# 使用例
result = handle_function_call(response)
if result:
    print(f"関数 {result['name']} を {result['args']} で実行")
else:
    print(f"テキスト応答: {response.text}")
```

### ベストプラクティス

Function Calling を効果的に使うためのポイントをまとめます。

**関数の説明は具体的に書く**: 「データを取得する」ではなく「商品データベースからキーワードで商品名・価格・在庫数を検索する」のように、何を・どこから・どんな形式で取得するかを明示してください。説明が曖昧だと、Gemini が適切なタイミングで関数を選択できなくなります。

**パラメータには型とデフォルト値を明記する**: `required` に含めるパラメータは最小限にし、オプションパラメータにはデフォルト値を説明に含めると、Gemini がより柔軟に関数を呼び出せます。

**エラー時は意味のあるメッセージを返す**: 関数の実行が失敗した場合、空の結果ではなくエラーの理由を Gemini に返してください。Gemini はエラー内容を理解し、ユーザーに適切な説明を行えます。

```

      ↕
```

## 1. 基本的な Function Calling

### Python 実装

```python
# function_calling_basic.py
import google.generativeai as genai
import json
from typing import Any

genai.configure(api_key="YOUR_API_KEY")

# ツール定義
WEATHER_TOOL = genai.protos.Tool(
    function_declarations=[
        genai.protos.FunctionDeclaration(
            name="get_weather",
            description="指定した都市の現在の天気を取得する",
            parameters=genai.protos.Schema(
                type=genai.protos.Type.OBJECT,
                properties={
                    "city": genai.protos.Schema(
                        type=genai.protos.Type.STRING,
                        description="都市名（例: Tokyo, Osaka）",
                    ),
                    "unit": genai.protos.Schema(
                        type=genai.protos.Type.STRING,
                        enum=["celsius", "fahrenheit"],
                        description="温度の単位",
                    ),
                },
                required=["city"],
            ),
        )
    ]
)

# ツール実装（実際はAPIを呼ぶ）
def get_weather(city: str, unit: str = "celsius") -> dict:
    # 本番では Open-Meteo や OpenWeatherMap API を呼ぶ
    mock_data = {
        "Tokyo": {"temp": 22, "condition": "晴れ", "humidity": 60},
        "Osaka": {"temp": 24, "condition": "曇り", "humidity": 65},
    }
    data = mock_data.get(city, {"temp": 20, "condition": "不明", "humidity": 50})
    temp = data["temp"]
    if unit == "fahrenheit":
        temp = temp * 9 / 5 + 32
    return {**data, "temp": temp, "unit": unit, "city": city}

# ツールディスパッチャー
TOOL_REGISTRY = {
    "get_weather": get_weather,
}

def dispatch_tool_call(function_call) -> str:
    func_name = function_call.name
    args = dict(function_call.args)

    if func_name not in TOOL_REGISTRY:
        return json.dumps({"error": f"未知のツール: {func_name}"})

    try:
        result = TOOL_REGISTRY[func_name](**args)
        return json.dumps(result, ensure_ascii=False)
    except Exception as e:
        return json.dumps({"error": str(e)})

# エージェントループ
def run_with_tools(user_message: str) -> str:
    model = genai.GenerativeModel(
        model_name="gemini-2.0-flash",
        tools=[WEATHER_TOOL],
    )

    chat = model.start_chat()
    response = chat.send_message(user_message)

    # ツール呼び出しループ
    while response.candidates[0].content.parts:
        parts = response.candidates[0].content.parts
        function_calls = [p for p in parts if hasattr(p, "function_call")]

        if not function_calls:
            break

        # ツール結果を返す
        tool_responses = []
        for part in function_calls:
            result = dispatch_tool_call(part.function_call)
            tool_responses.append(
                genai.protos.Part(
                    function_response=genai.protos.FunctionResponse(
                        name=part.function_call.name,
                        response={"result": json.loads(result)},
                    )
                )
            )

        response = chat.send_message(tool_responses)

    return response.text

# 使用例
if __name__ == "__main__":
    result = run_with_tools("東京と大阪の今日の天気を比較してください")
    print(result)
```

## 3. 並列 Function Calling（高度な実装）

```python
# parallel_function_calling.py
import asyncio
import json
import google.generativeai as genai
from google.generativeai.types import FunctionLibrary

# 複数ツールの並列実行をサポートするエージェント
class ParallelToolAgent:
    def __init__(self, tools: list, model_name: str = "gemini-2.0-flash"):
        self.model = genai.GenerativeModel(
            model_name=model_name,
            tools=tools,
            tool_config={"function_calling_config": {"mode": "AUTO"}},
        )
        self.tool_registry: dict = {}

    def register_tool(self, name: str, func):
        self.tool_registry[name] = func

    async def _call_tool_async(self, function_call) -> genai.protos.Part:
        """非同期でツールを実行"""
        func_name = function_call.name
        args = dict(function_call.args)

        if func_name not in self.tool_registry:
            result = {"error": f"未知のツール: {func_name}"}
        else:
            try:
                func = self.tool_registry[func_name]
                # async 関数と同期関数の両方をサポート
                if asyncio.iscoroutinefunction(func):
                    result = await func(**args)
                else:
                    result = await asyncio.get_event_loop().run_in_executor(
                        None, lambda: func(**args)
                    )
            except Exception as e:
                result = {"error": str(e)}

        return genai.protos.Part(
            function_response=genai.protos.FunctionResponse(
                name=func_name,
                response={"result": result},
            )
        )

    async def run(self, user_message: str, max_iterations: int = 10) -> str:
        chat = self.model.start_chat()
        response = await asyncio.get_event_loop().run_in_executor(
            None, lambda: chat.send_message(user_message)
        )

        for _ in range(max_iterations):
            parts = response.candidates[0].content.parts
            function_calls = [p for p in parts if hasattr(p, "function_call") and p.function_call.name]

            if not function_calls:
                break

            # 全ツール呼び出しを並列実行 ← ここがポイント
            tool_results = await asyncio.gather(
                *[self._call_tool_async(p.function_call) for p in function_calls]
            )

            response = await asyncio.get_event_loop().run_in_executor(
                None, lambda: chat.send_message(list(tool_results))
            )

        return response.text
```

## 4. TypeScript 実装（Next.js API Route 版）

```typescript
// app/api/gemini-agent/route.ts
import { GoogleGenerativeAI, FunctionCallingMode } from '@google/generative-ai';
import { NextRequest, NextResponse } from 'next/server';

const genai = new GoogleGenerativeAI(process.env.GOOGLE_API_KEY!);

// ツール定義
const TOOLS = [
  {
    functionDeclarations: [
      {
        name: 'search_products',
        description: '商品を検索する',
        parameters: {
          type: 'object' as const,
          properties: {
            query: { type: 'string', description: '検索クエリ' },
            maxResults: { type: 'number', description: '最大件数（デフォルト5）' },
          },
          required: ['query'],
        },
      },
      {
        name: 'get_product_details',
        description: '商品の詳細情報を取得する',
        parameters: {
          type: 'object' as const,
          properties: {
            productId: { type: 'string', description: '商品ID' },
          },
          required: ['productId'],
        },
      },
    ],
  },
];

// ツール実装
async function searchProducts(query: string, maxResults = 5): Promise<object> {
  // 実際はデータベースを検索
  return {
    results: Array.from({ length: maxResults }, (_, i) => ({
      id: `prod_${i + 1}`,
      name: `${query} 商品 ${i + 1}`,
      price: (1000 + i * 500),
    })),
  };
}

async function getProductDetails(productId: string): Promise<object> {
  return {
    id: productId,
    name: `商品 ${productId}`,
    price: 1500,
    description: '詳細な商品説明...',
    stock: 42,
  };
}

const TOOL_DISPATCH: Record<string, (...args: any[]) => Promise<object>> = {
  search_products: ({ query, maxResults }: { query: string; maxResults?: number }) =>
    searchProducts(query, maxResults),
  get_product_details: ({ productId }: { productId: string }) =>
    getProductDetails(productId),
};

export async function POST(req: NextRequest) {
  const { message } = await req.json();

  const model = genai.getGenerativeModel({
    model: 'gemini-2.0-flash',
    tools: TOOLS,
    toolConfig: { functionCallingConfig: { mode: FunctionCallingMode.AUTO } },
  });

  const chat = model.startChat();
  let response = await chat.sendMessage(message);

  // ツールループ
  for (let i = 0; i < 10; i++) {
    const functionCalls = response.response.functionCalls();
    if (!functionCalls || functionCalls.length === 0) break;

    // 並列実行
    const results = await Promise.all(
      functionCalls.map(async (fc) => {
        const func = TOOL_DISPATCH[fc.name];
        const result = func ? await func(fc.args as any) : { error: `未知のツール: ${fc.name}` };
        return { functionResponse: { name: fc.name, response: result } };
      }),
    );

    response = await chat.sendMessage(results);
  }

  return NextResponse.json({ result: response.response.text() });
}
```
