Googleドライブにファイルが増え続けて「どこに何があるかわからない」という経験はないでしょうか。週報、企画書、議事録……気づけば整理されないまま数百件を超えていることもあります。

## Google Drive API との連携：ファイルを取得する

まず、GoogleドライブからドキュメントファイルのリストをPythonで取得する関数を作ります。

```python
import os
import io
from google.oauth2.credentials import Credentials
from google_auth_oauthlib.flow import InstalledAppFlow
from google.auth.transport.requests import Request
from googleapiclient.discovery import build
from googleapiclient.http import MediaIoBaseDownload

# 必要なOAuthスコープ
SCOPES = [
    'https://www.googleapis.com/auth/drive.readonly',
    'https://www.googleapis.com/auth/drive.file'
]

def get_drive_service():
    """Google Drive APIサービスを初期化して返す"""
    creds = None
    # token.json にキャッシュされた認証情報があれば利用する
    if os.path.exists('token.json'):
        creds = Credentials.from_authorized_user_file('token.json', SCOPES)
    if not creds or not creds.valid:
        if creds and creds.expired and creds.refresh_token:
            creds.refresh(Request())
        else:
            flow = InstalledAppFlow.from_client_secrets_file('credentials.json', SCOPES)
            creds = flow.run_local_server(port=0)
        with open('token.json', 'w') as token:
            token.write(creds.to_json())
    return build('drive', 'v3', credentials=creds)

def list_documents(service, folder_id=None, max_files=20):
    """Google Driveのドキュメントファイル一覧を取得する"""
    query = "mimeType='application/vnd.google-apps.document' and trashed=false"
    if folder_id:
        query += f" and '{folder_id}' in parents"
    results = service.files().list(
        q=query,
        pageSize=max_files,
        fields="files(id, name, mimeType, createdTime)"
    ).execute()
    return results.get('files', [])

def export_document_as_text(service, file_id):
    """Google Documentをプレーンテキストとしてエクスポートする"""
    request = service.files().export_media(
        fileId=file_id,
        mimeType='text/plain'
    )
    buffer = io.BytesIO()
    downloader = MediaIoBaseDownload(buffer, request)
    done = False
    while not done:
        _, done = downloader.next_chunk()
    return buffer.getvalue().decode('utf-8', errors='replace')
```

## Gemini API でファイルを分析・分類する

次に、取得したテキストをGemini APIに渡して、内容の要約とカテゴリ分類を行います。

```python
import google.generativeai as genai

def analyze_document(text: str, filename: str) -> dict:
    """
    Gemini APIでドキュメントを要約・分類する
    戻り値: {"summary": str, "category": str, "keywords": list}
    """
    genai.configure(api_key=os.environ["GEMINI_API_KEY"])
    model = genai.GenerativeModel("gemini-2.0-flash")

    # テキストが長すぎる場合は先頭2000文字に制限する
    truncated = text[:2000]

    prompt = f"""以下のビジネスドキュメントを分析してください。

ファイル名: {filename}
内容（先頭部分）:
{truncated}

以下の形式でJSONを返してください:
{{
  "summary": "ドキュメントの内容を2〜3文で要約",
  "category": "次の中から最も適切な1つ: 議事録/企画書/報告書/マニュアル/メモ/その他",
  "keywords": ["キーワード1", "キーワード2", "キーワード3"]
}}
JSONのみ返してください。説明文は不要です。"""

    response = model.generate_content(prompt)
    import json, re
    # JSONブロックを抽出してパースする
    json_match = re.search(r'\{.*\}', response.text, re.DOTALL)
    if json_match:
        return json.loads(json_match.group())
    return {"summary": "解析不可", "category": "その他", "keywords": []}
```

このコードでは `gemini-2.0-flash` モデルを使っています。高速で費用も抑えられるため、バッチ処理に向いています。分析対象のテキストが長い場合は先頭2,000文字に限定していますが、Gemini APIは最大100万トークンのコンテキストウィンドウを持つため、より精度を上げたい場合はテキスト量を増やすことも可能です。

また、Gemini APIの構造化出力（Structured Output）機能を使うと、JSONのパースをより確実に行えます。詳しい活用方法については、[Gemini API × ECサイト自動化 商品コンテンツAI生成ガイド](/articles/gemini-api/gemini-api-ecommerce-product-content-automation) でも解説しています。

## 自動整理ワークフロー：分類→フォルダ移動

分析結果をもとに、ファイルを対応フォルダへ自動移動する仕組みを実装します。

```python
def get_or_create_folder(service, folder_name: str, parent_id: str = None) -> str:
    """指定した名前のフォルダを取得または作成してIDを返す"""
    query = f"name='{folder_name}' and mimeType='application/vnd.google-apps.folder' and trashed=false"
    if parent_id:
        query += f" and '{parent_id}' in parents"
    results = service.files().list(q=query, fields="files(id)").execute()
    folders = results.get('files', [])
    if folders:
        return folders[0]['id']
    # フォルダが存在しなければ作成する
    metadata = {
        'name': folder_name,
        'mimeType': 'application/vnd.google-apps.folder'
    }
    if parent_id:
        metadata['parents'] = [parent_id]
    folder = service.files().create(body=metadata, fields='id').execute()
    return folder['id']

def move_file_to_folder(service, file_id: str, new_folder_id: str):
    """ファイルを指定フォルダに移動する"""
    # 現在の親フォルダを取得してから移動する
    file = service.files().get(fileId=file_id, fields='parents').execute()
    previous_parents = ','.join(file.get('parents', []))
    service.files().update(
        fileId=file_id,
        addParents=new_folder_id,
        removeParents=previous_parents,
        fields='id, parents'
    ).execute()

def organize_drive(root_folder_id: str = None, dry_run: bool = True):
    """
    Google Driveのドキュメントを自動整理するメイン関数
    dry_run=True の場合は実際の移動を行わず確認のみ
    """
    service = get_drive_service()
    files = list_documents(service, folder_id=root_folder_id, max_files=10)

    print(f"対象ファイル数: {len(files)}\n")

    for file in files:
        print(f"処理中: {file['name']}")
        # ドキュメントのテキストを取得する
        try:
            text = export_document_as_text(service, file['id'])
        except Exception as e:
            print(f"  ⚠️ テキスト取得失敗: {e}")
            continue

        # Gemini APIで分析する
        result = analyze_document(text, file['name'])
        category = result.get('category', 'その他')
        summary = result.get('summary', '')

        print(f"  カテゴリ: {category}")
        print(f"  要約: {summary}")

        if not dry_run:
            # カテゴリフォルダを作成して移動する
            folder_id = get_or_create_folder(service, category, parent_id=root_folder_id)
            move_file_to_folder(service, file['id'], folder_id)
            print(f"  ✅ 「{category}」フォルダに移動しました")
        else:
            print(f"  [DRY RUN] 「{category}」フォルダへ移動予定")
        print()

if __name__ == "__main__":
    # まず dry_run=True で動作確認してから False に変更する
    organize_drive(dry_run=True)
```

実行前は必ず `dry_run=True` で確認してください。意図しないファイル移動を防ぐためのセーフガードです。

## よくあるエラーと対処法

実装中に遭遇しやすいエラーと解決策を紹介します。

**`HttpError 403: The caller does not have permission`**
Google Drive APIのスコープが不足しています。`SCOPES` に `drive.file` だけでなく `drive` または `drive.readonly` を追加してください。`token.json` を削除して再認証が必要です。

**`google.auth.exceptions.RefreshError`**
OAuthトークンの有効期限が切れています。`token.json` を削除して再度 `organize_drive()` を実行し、ブラウザで再認証してください。

**Gemini APIのレスポンスがJSONでない**
プロンプトに「JSONのみ返してください」と明示しているにもかかわらず説明文が混入する場合があります。`re.search(r'\{.*\}', ..., re.DOTALL)` で確実に抽出するか、Gemini APIの `response_mime_type="application/json"` オプションを使うと安定します。

**処理速度が遅い**
Gemini APIは無料枠の場合、1分あたりのリクエスト数（RPM）に制限があります。ファイルが多い場合は `time.sleep(1)` を各ループに追加して制限に引っかからないようにしてください。

## FAQ

**Q: Google Workspaceの有料プランが必要ですか？**
A: いいえ。Google Drive APIは無料の個人アカウントでも利用可能です。ただし、Google Cloud Console でプロジェクトを作成してOAuth認証情報を設定する必要があります。

**Q: PDFやWord形式のファイルにも対応していますか？**
A: このガイドでは Google Docsを対象にしていますが、PDFの場合は `mimeType='application/pdf'` をクエリに追加し、File APIを使ってGemini APIに直接渡す方法もあります。詳しくは [Gemini API × PDFドキュメント解析ガイド](/articles/gemini-api/gemini-api-pdf-analysis-guide) をご参照ください。

**Q: 処理費用はどのくらいかかりますか？**
A: `gemini-2.0-flash` を使う場合、100万トークン入力あたり約 $0.075（2026年4月時点）です。1ファイルを2,000文字で処理すると概算で約1,000トークンですので、100ファイル処理してもごく少額で済みます。API料金の詳細な最適化方法については [Gemini APIコスト完全最適化ガイド](/articles/gemini-api/gemini-api-cost-optimization-complete-guide) も参考にしてください。

**Q: 一度に大量のファイルを処理できますか？**
A: 技術的には可能ですが、無料枠のレート制限（RPMやTPM）に注意が必要です。本記事では `max_files=10` で始めることを推奨しています。商用利用や大量処理を予定している場合は、Vertex AI経由での利用を検討してください。

**Q: 分類のカテゴリをカスタマイズできますか？**
A: はい。`analyze_document` 関数内のプロンプトで「次の中から最も適切な1つ」として指定しているカテゴリ名を、業務に合わせて自由に変更できます。例えば「採用関連/財務/マーケティング/技術仕様」など、組織ごとのカテゴリに対応可能です。
