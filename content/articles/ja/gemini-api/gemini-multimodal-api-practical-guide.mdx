## 画像の最適化

大規模な画像処理ではサイズと品質のバランスが重要です。

```javascript
const sharp = require("sharp");

class ImageOptimizer {
  async optimizeForGemini(inputPath, outputPath) {
    // 最適な解像度: 4096x4096以下
    // ファイルサイズ: 20MB以下
    await sharp(inputPath)
      .resize(2048, 2048, {
        fit: "inside",
        withoutEnlargement: true
      })
      .jpeg({ quality: 85 })
      .toFile(outputPath);
  }
  
  async optimizeBatch(inputDirectory, outputDirectory) {
    const fs = require("fs");
    const files = fs.readdirSync(inputDirectory);
    
    for (const file of files) {
      const inputPath = `${inputDirectory}/${file}`;
      const outputPath = `${outputDirectory}/${file}`;
      
      await this.optimizeForGemini(inputPath, outputPath);
    }
  }
}
```

## エラーハンドリングと再試行

```javascript
class RobustGeminiClient {
  constructor(maxRetries = 3) {
    this.maxRetries = maxRetries;
  }
  
  async analyzeWithRetry(imagePath, prompt) {
    let lastError;
    
    for (let attempt = 1; attempt <= this.maxRetries; attempt++) {
      try {
        const fs = require("fs");
        const imageData = fs.readFileSync(imagePath);
        const base64Image = imageData.toString("base64");
        
        const response = await client.messages.create({
          model: "gemini-2.0-flash",
          max_tokens: 1024,
          messages: [
            {
              role: "user",
              content: [
                {
                  type: "image",
                  source: {
                    type: "base64",
                    media_type: "image/jpeg",
                    data: base64Image
                  }
                },
                {
                  type: "text",
                  text: prompt
                }
              ]
            }
          ]
        });
        
        return {
          success: true,
          content: response.content[0].text,
          attempt: attempt
        };
        
      } catch (error) {
        lastError = error;
        
        if (error.status === 429) {
          // Rate Limit
          const waitTime = 1000 * Math.pow(2, attempt - 1);
          console.log(`Rate limited. Waiting ${waitTime}ms`);
          await new Promise(r => setTimeout(r, waitTime));
          continue;
        } else if (error.status === 500 || error.status === 503) {
          // Server Error
          console.log(`Server error. Retrying`);
          await new Promise(r => setTimeout(r, 1000 * attempt));
          continue;
        } else if (error.status === 400) {
          // Bad Request - don't retry
          throw error;
        }
      }
    }
    
    return {
      success: false,
      error: lastError.message,
      attempts: this.maxRetries
    };
  }
}
```

## コスト最適化

```javascript
class CostOptimizedAnalyzer {
  constructor(imagePricing = 0.001, outputPricing = 0.0001) {
    this.imagePricing = imagePricing;
    this.outputPricing = outputPricing;
    this.totalCost = 0;
  }
  
  estimateCost(imageCount, averageOutputTokens) {
    const imageCost = imageCount * this.imagePricing;
    const outputCost = averageOutputTokens * this.outputPricing;
    return imageCost + outputCost;
  }
  
  async analyzeWithCostTracking(imagePath, prompt) {
    const estimatedCost = this.estimateCost(1, 500);
    
    console.log(`Estimated cost: $${estimatedCost.toFixed(4)}`);
    
    const result = await client.messages.create({
      model: "gemini-2.0-flash",
      max_tokens: 1024,
      messages: [/* ... */]
    });
    
    // 実際のコスト計算
    const actualCost = estimatedCost; // 簡略化
    this.totalCost += actualCost;
    
    return {
      result: result.content[0].text,
      cost: actualCost.toFixed(4),
      totalCost: this.totalCost.toFixed(2)
    };
  }
}
```

## 実装統合: プロダクション対応分析システム

```javascript
class ProductionGeminiAnalyzer {
  constructor(config = {}) {
    this.optimizer = new ImageOptimizer();
    this.analyzer = new MultimodalAnalyzer();
    this.costTracker = new CostOptimizedAnalyzer();
    this.maxRetries = config.maxRetries || 3;
  }
  
  async analyzeProduct(productImage, productDescription) {
    try {
      // 1. 画像最適化
      const optimizedImage = await this.optimizer.optimizeForGemini(productImage, "/tmp/optimized.jpg");
      
      // 2. 分析実行
      const analysis = await this.analyzer.analyzeProductImage(optimizedImage, productDescription);
      
      // 3. コスト追跡
      const result = await this.costTracker.analyzeWithCostTracking(optimizedImage, productDescription);
      
      return {
        success: true,
        analysis: analysis,
        cost: result.cost,
        totalCost: result.totalCost
      };
      
    } catch (error) {
      console.error("Analysis failed:", error.message);
      return {
        success: false,
        error: error.message
      };
    }
  }
}
```

## 全体を振り返って

Gemini APIのマルチモーダル機能活用の鍵：

1. **画像最適化は必須** — サイズと品質のバランスを取る
2. **プロンプト設計が精度を左右** — 詳細で構造化された指示を心がける
3. **エラーハンドリングは本番必須** — Rate Limitやサーバーエラーへの対応
4. **コスト管理が重要** — 画像数と出力トークンをトラッキング
5. **複数モーダルの組み合わせ** — 画像だけでなくテキストとの連携が効果的

これらのパターンを実装することで、実用的で信頼性高いマルチモーダル分析システムが構築できます。
