前編で「コスト設計は事業開始前に決めるべき」という話をしました。では、その設計をどう実装するのか。

個人開発者が「手作り SaaS」を作る際、サーバーレスアーキテクチャは必須です。管理負担が少なく、スケールに応じて自動的にコストも増える。失敗しても月数千円の損失で済みます。

このガイドで解説するのは：

- **Next.js フロントエンド** — Stripe Checkout 統合
- **Cloud Run バックエンド** — Gemini API 呼び出し + トークン計測
- **Firestore** — ユーザー・セッション・利用量の永続化
- **Stripe Metered Billing** — 月末精算自動化
- **Context Caching + Batch API** — コスト削減テクニック

実装したら、月額原価は ¥3,000 以下。100 ユーザーで月 50 万円以上の売上を見込める構造です。

## アーキテクチャ全体図

```
[Next.js Frontend]
    ↓
    ├→ Stripe Checkout（サブスク + メタ課金）
    ├→ Cloud Run API（Gemini 呼び出し）
    └→ Firestore（セッション・利用量記録）

[Cloud Run Backend]
    ├→ Gemini API（チャット実行）
    ├→ Stripe Metering API（月末精算用の利用量送信）
    └→ Context Caching + Batch API（コスト最適化）
```

## Step 2: Cloud Run での Gemini API 呼び出し実装

Node.js + Express での実装：

```javascript
const express = require("express");
const admin = require("firebase-admin");
const { GoogleGenerativeAI } = require("@google/generative-ai");
const stripe = require("stripe")(process.env.STRIPE_SECRET_KEY);

const app = express();
const db = admin.firestore();
const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY);

// Gemini API 呼び出しエンドポイント
app.post("/api/chat", async (req, res) => {
  const { userId, message, conversationHistory, useCaching } = req.body;

  try {
    // 1. ユーザーを Firestore から取得
    const userDoc = await db.collection("users").doc(userId).get();
    if (!userDoc.exists) {
      return res.status(404).json({ error: "User not found" });
    }

    const userData = userDoc.data();

    // 2. プランの利用制限チェック
    if (userData.plan === "free" && userData.usage.tokensThisMonth > 50000) {
      return res.status(429).json({ 
        error: "Free plan monthly limit exceeded",
        tokensUsed: userData.usage.tokensThisMonth
      });
    }

    // 3. システムプロンプト（Context Caching の対象）
    const systemPrompt = `You are a helpful AI assistant. 
Respond in Japanese. Keep responses concise and actionable.
Current plan: ${userData.plan}
Last reset: ${userData.usage.lastResetAt}`;

    // 4. Gemini API を Context Caching 付きで呼び出す
    const model = genAI.getGenerativeModel({
      model: "gemini-2.5-pro"
    });

    const cachingConfig = useCaching ? {
      cachedContent: {
        systemInstruction: {
          parts: [{ text: systemPrompt }],
          role: "user"
        },
        cachedContentExpiration: {
          ttlSeconds: 300  // 5 分有効
        }
      }
    } : {};

    const response = await model.generateContent({
      ...cachingConfig,
      contents: [
        ...conversationHistory.map(msg => ({
          role: msg.role,
          parts: [{ text: msg.content }]
        })),
        {
          role: "user",
          parts: [{ text: message }]
        }
      ]
    });

    const generatedText = response.text;

    // 5. トークン使用量を推定（実際の API レスポンスを待つ）
    const inputTokens = Math.ceil(
      (systemPrompt.length + JSON.stringify(conversationHistory).length) / 4
    );
    const outputTokens = Math.ceil(generatedText.length / 4);
    const totalTokens = inputTokens + outputTokens;

    // 6. 利用量を Firestore に記録
    const logId = `${userId}-${Date.now()}`;
    await db.collection("usage-logs").doc(logId).set({
      userId,
      timestamp: admin.firestore.Timestamp.now(),
      requestId: `req-${Date.now()}`,
      inputTokens,
      outputTokens,
      totalTokens,
      costUSD: (inputTokens * 0.0015 + outputTokens * 0.006) / 1000,
      model: "gemini-2.5-pro",
      cached: useCaching
    });

    // 7. ユーザーの月間利用量を更新
    await db.collection("users").doc(userId).update({
      "usage.tokensThisMonth": admin.firestore.FieldValue.increment(totalTokens),
      "usage.requestsThisMonth": admin.firestore.FieldValue.increment(1)
    });

    // 8. 月末精算用に Stripe Metering API へ送信
    // （以下の Step 3 で詳細説明）
    await sendToStripeMeteringAPI(userId, totalTokens, inputTokens, outputTokens);

    res.json({
      response: generatedText,
      tokensUsed: totalTokens,
      estimatedCostUSD: (inputTokens * 0.0015 + outputTokens * 0.006) / 1000,
      monthlyTotalTokens: userData.usage.tokensThisMonth + totalTokens
    });

  } catch (error) {
    console.error("API Error:", error);
    res.status(500).json({ error: error.message });
  }
});

app.listen(8080, () => console.log("Cloud Run listening on port 8080"));
```

## Step 3: Stripe Metered Billing の統合

Stripe Metered Billing により、API 利用量に応じた月末精算が自動化されます：

```javascript
// Stripe Metering API へトークン使用量を送信
async function sendToStripeMeteringAPI(userId, totalTokens, inputTokens, outputTokens) {
  const userDoc = await db.collection("users").doc(userId).get();
  const userData = userDoc.data();

  // Stripe Subscription の meterEvent を登録
  const planToPricingId = {
    "pro": process.env.STRIPE_PRO_METERED_PRICE_ID,
    "premium": process.env.STRIPE_PREMIUM_METERED_PRICE_ID
  };

  const priceId = planToPricingId[userData.plan];
  if (!priceId) return;  // Free プランはメタ課金しない

  // 1 リクエスト = X 単位の課金という設定の場合
  // 例：1,000 トークン = 0.1 単位（最小 $0.0015）
  const meterUnits = Math.max(0.0001, totalTokens / 10000);

  try {
    await stripe.billing.meterEvents.create({
      event_name: "gemini_tokens",
      timestamp: Math.floor(Date.now() / 1000),
      customer: userData.stripeCustomerId,
      value: meterUnits
    });
  } catch (error) {
    console.error("Stripe Metering API Error:", error);
  }
}
```

## Step 4: Context Caching を活用した原価削減

同じユーザーが繰り返し使う場合、Context Caching が大きな効果を発揮します：

```javascript
// ユーザーのセッションごとに Context Cache を保持
const cacheStore = new Map();  // userId -> { contentHash, expiresAt }

async function getCachedContent(userId) {
  const cached = cacheStore.get(userId);
  
  if (cached && cached.expiresAt > Date.now()) {
    return cached.contentHash;
  }

  // キャッシュ無効 → 新しいシステムプロンプトでキャッシュ作成
  const systemPrompt = generateSystemPrompt(userId);
  const contentHash = crypto
    .createHash("sha256")
    .update(systemPrompt)
    .digest("hex");

  cacheStore.set(userId, {
    contentHash,
    expiresAt: Date.now() + 300000  // 5 分
  });

  return contentHash;
}

// API 呼び出し時
async function callGeminiWithCache(userId, message, conversationHistory) {
  const cachedContentHash = await getCachedContent(userId);

    model: "gemini-2.5-pro",

      temperature: 1,
      topP: 0.95,
      topK: 40
    }
  }).generateContent({
    cachedContent: {
      // ... 前述の実装参照
    }
  });
```

## Step 5: Batch API の活用シーン

リアルタイムチャットではなく「バッチ分析」の場合、Batch API で 50% コスト削減：

```javascript
// 定期的な分析タスク（Cloud Scheduler + Cloud Functions）
async function batchAnalyzeUserContent(userId) {
  const userDocs = await db.collection("users")
    .where("plan", "in", ["pro", "premium"])
    .limit(100)
    .get();

  const batchRequests = [];

  userDocs.forEach(doc => {
    const userData = doc.data();
    batchRequests.push({
      custom_id: `${userData.userId}-analysis-${Date.now()}`,
      params: {
        model: "gemini-2.5-pro",
        systemInstruction: "Analyze the provided text and return key insights.",
        contents: [{
          role: "user",
          parts: [{ text: userData.contentToAnalyze }]
        }]
      }
    });
  });

  // Google AI Batch Processing にリクエスト送信
  const batchJob = await genAI.batchProcessing({
    requests: batchRequests
  });

  console.log(`Batch job ${batchJob.id} submitted. Results arrive in 24h.`);

  // 結果は webhook で受け取り Firestore に保存
  // → ユーザーへメール通知
}
```

## Step 6: 月額費用の監視ダッシュボード

運用開始後、毎日の API 費用を監視することが重要です：

```javascript
// Cloud Run 定期実行（毎日深夜）
async function calculateDailyMetrics() {
  const today = new Date();
  today.setHours(0, 0, 0, 0);

  const logsToday = await db.collection("usage-logs")
    .where("timestamp", ">=", admin.firestore.Timestamp.fromDate(today))
    .get();

  let totalTokens = 0;
  let totalCostUSD = 0;
  let requestCount = 0;
  const userBreakdown = {};

  logsToday.forEach(doc => {
    const log = doc.data();
    totalTokens += log.totalTokens;
    totalCostUSD += log.costUSD;
    requestCount += 1;

    if (!userBreakdown[log.userId]) {
      userBreakdown[log.userId] = { tokens: 0, cost: 0, requests: 0 };
    }
    userBreakdown[log.userId].tokens += log.totalTokens;
    userBreakdown[log.userId].cost += log.costUSD;
    userBreakdown[log.userId].requests += 1;
  });

  // メトリクスを Firestore に保存（ダッシュボード表示用）
  await db.collection("metrics").doc(`daily-${today.toISOString().split("T")[0]}`).set({
    date: today,
    totalTokens,
    totalCostUSD,
    requestCount,
    averageCostPerRequest: totalCostUSD / requestCount,
    userBreakdown,
    projectedMonthlyCostUSD: totalCostUSD * 30
  });

  console.log(`Daily: ${totalTokens} tokens, $${totalCostUSD.toFixed(2)}, Projected monthly: $${(totalCostUSD * 30).toFixed(2)}`);
}
```

## Step 7: エラー処理とレート制限

実運用では以下の問題が必ず起きます：

```javascript
// Gemini API レート制限（429 Too Many Requests）への対応
async function callGeminiWithRetry(userId, message, maxRetries = 3) {
  for (let attempt = 0; attempt < maxRetries; attempt++) {
    try {
      return await model.generateContent({ /* ... */ });
    } catch (error) {
      if (error.status === 429) {
        // Exponential backoff
        const waitMs = Math.pow(2, attempt) * 1000 + Math.random() * 1000;
        console.log(`Rate limited. Waiting ${waitMs}ms before retry ${attempt + 1}...`);
        await new Promise(resolve => setTimeout(resolve, waitMs));
        continue;
      }
      throw error;
    }
  }
  throw new Error("Max retries exceeded");
}

// Stripe メタ課金失敗時のリトライ
async function sendToStripeMeteringAPIWithFallback(userId, totalTokens, inputTokens, outputTokens) {
  try {
    await sendToStripeMeteringAPI(userId, totalTokens, inputTokens, outputTokens);
  } catch (error) {
    console.error("Stripe metering failed, storing locally:", error);
    
    // ローカルに待機中のメタイベントを保存
    await db.collection("pending-meter-events").add({
      userId,
      totalTokens,
      inputTokens,
      outputTokens,
      timestamp: admin.firestore.Timestamp.now(),
      retryCount: 0
    });

    // 別の定期ジョブでリトライ
  }
}
```

### 問題パターンと対処

**1. 特定ユーザーが異常な利用量**
→ キャッシュが有効になっていない可能性。セッション設計を見直す

**2. Batch API の結果が来ない**
→ 24 時間以上待つ。webhooks がセットアップされているか確認

**3. Stripe と Gemini のトークン計算がズレている**
→ 実際の API リスポンス（usageMetadata）と推定値を照合。修正係数を適用
