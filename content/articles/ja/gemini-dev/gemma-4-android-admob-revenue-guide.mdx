### Step 4: Gemma推論エンジンInitialize

Kotlinコード：

```kotlin
// GemmaEngine.kt
import com.google.mediapipe.tasks.genai.llminference.LlmInference

class GemmaEngine(context: Context) {
    private var llmInference: LlmInference? = null
    
    init {
        val options = LlmInference.LlmInferenceOptions.builder()
            .setModelAssetPath("gemma-2b-it-gpu-android.task")
            .setMaxTokens(512)
            .setTemperature(0.7f)
            .setTopK(40)
            .setTopP(0.95f)
            .build()
        
        llmInference = LlmInference.createFromOptions(context, options)
    }
    
    fun generateResponse(prompt: String): String {
        return try {
            val response = llmInference?.generateResponse(prompt)
            response ?: "Error: No response"
        } catch (e: Exception) {
            "Error: ${e.message}"
        }
    }
    
    fun close() {
        llmInference?.close()
    }
}
```

### Step 5: UI統合（Jetpack Compose）

```kotlin
// MainActivity.kt
import androidx.compose.foundation.layout.*
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import kotlinx.coroutines.launch

class ChatViewModel(private val engine: GemmaEngine) : ViewModel() {
    var userInput by mutableStateOf("")
    var chatHistory by mutableStateOf<List<Pair<String, String>>>(emptyList())
    var isLoading by mutableStateOf(false)
    
    fun sendMessage(text: String) {
        if (text.isBlank()) return
        
        isLoading = true
        viewModelScope.launch {
            val response = engine.generateResponse(text)
            chatHistory = chatHistory + (text to response)
            userInput = ""
            isLoading = false
        }
    }
}

@Composable
fun ChatScreen(viewModel: ChatViewModel) {
    Column(modifier = Modifier
        .fillMaxSize()
        .padding(16.dp)) {
        
        // Chat history
        LazyColumn(modifier = Modifier
            .weight(1f)
            .fillMaxWidth()) {
            items(viewModel.chatHistory.size) { index ->
                val (user, bot) = viewModel.chatHistory[index]
                
                ChatBubble(
                    text = user,
                    isUser = true,
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(vertical = 8.dp)
                )
                
                ChatBubble(
                    text = bot,
                    isUser = false,
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(vertical = 8.dp)
                )
            }
        }
        
        // Input bar
        Row(modifier = Modifier
            .fillMaxWidth()
            .padding(vertical = 16.dp),
            horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            
            TextField(
                value = viewModel.userInput,
                onValueChange = { viewModel.userInput = it },
                modifier = Modifier
                    .weight(1f)
                    .height(56.dp),
                placeholder = { Text("Message...") },
                enabled = \!viewModel.isLoading
            )
            
            Button(
                onClick = { viewModel.sendMessage(viewModel.userInput) },
                enabled = \!viewModel.isLoading,
                modifier = Modifier.height(56.dp)
            ) {
                Text(if (viewModel.isLoading) "..." else "Send")
            }
        }
    }
}

@Composable
fun ChatBubble(text: String, isUser: Boolean, modifier: Modifier = Modifier) {
    Card(modifier = modifier
        .widthIn(max = 300.dp),
        colors = CardDefaults.cardColors(
            containerColor = if (isUser) Color.Blue else Color.Gray
        )) {
        Text(text, modifier = Modifier.padding(12.dp), color = Color.White)
    }
```

```kotlin
import com.google.firebase.analytics.FirebaseAnalytics

class AnalyticsManager(context: Context) {
    private val firebaseAnalytics = FirebaseAnalytics.getInstance(context)

        firebaseAnalytics.logEvent("ai_response") {
            param("prompt_length", prompt.length.toLong())
            param("response_length", responseLength.toLong())
            param("user_level", userLevel)
            param("timestamp", System.currentTimeMillis())
        }
    }
    
    fun logAdImpression(adFormat: String, placement: String) {
        firebaseAnalytics.logEvent("ad_impression") {
            param("format", adFormat)  // banner, interstitial, rewarded
            param("placement", placement)  // top, middle, bottom
            param("user_segment", "TBD")
        }
    }
    
    fun logSessionMetrics(duration: Long, messageCount: Int) {
        firebaseAnalytics.logEvent("session_end") {
            param("duration_seconds", (duration / 1000).toLong())
            param("message_count", messageCount.toLong())
        }
    }
}
```
