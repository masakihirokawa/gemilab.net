## What Is Lyria 3 Pro?

Lyria is Google DeepMind's music generation AI. Lyria 3 Pro represents a generational leap over previous versions:

- **Up to 3 minutes per generation**: Previous versions topped out at 30–60 seconds
- **Lyrics included**: Generates multilingual vocals alongside the music — English, Japanese, and more
- **Auto-structured songs**: Intro → verse → chorus → bridge — Lyria handles the architecture
- **Image-to-music generation**: Feed it a photo, and it generates music that matches the mood
- **Fine-grained style control**: Specify genre, tempo, instrumentation, and emotional tone

Lyria 3 Pro is included in the **Google AI Ultra** and **Google AI Pro** subscriptions at no extra charge.

## API Integration

Here's how to integrate Lyria 3 Pro into your projects:

```python
import google.generativeai as genai
import base64

# Configure your API key
genai.configure(api_key="YOUR_GEMINI_API_KEY")

def generate_music(prompt: str, output_path: str = "output.mp3") -> str:
    """
    Generate music with Lyria 3 Pro and save to file.

    Args:
        prompt: Music generation prompt (English or multilingual)
        output_path: File path for the saved audio

    Returns:
        Path to the saved audio file
    """
    model = genai.GenerativeModel('lyria-3-pro')

    response = model.generate_content(
        prompt,
        generation_config={'audio_timestamp': True}
    )

    # Extract and save audio data
    for part in response.candidates[0].content.parts:
        if hasattr(part, 'inline_data') and part.inline_data.mime_type.startswith('audio'):
            audio_data = base64.b64decode(part.inline_data.data)
            with open(output_path, 'wb') as f:
                f.write(audio_data)
            print(f"✅ Saved to: {output_path}")
            return output_path

    raise ValueError("No audio data in response")


# Example usage
prompt = """
An acoustic singer-songwriter song about finding courage to start over.
Gentle fingerpicked guitar, soft piano, warm vocals.
Tempo: 75 BPM. English lyrics. 2 minutes 30 seconds.
Verse-chorus-verse-chorus-bridge-chorus structure.
"""
generate_music(prompt, "new_beginning.mp3")
```

### Image-to-Music Generation

One of Lyria 3 Pro's most creative features — generate music that matches the mood of a photo:

```python
import google.generativeai as genai
import base64

genai.configure(api_key="YOUR_GEMINI_API_KEY")

def generate_music_from_image(
    image_path: str,
    style_hint: str = "",
    output_path: str = "from_image.mp3"
) -> str:
    """
    Generate music inspired by an image.

    Args:
        image_path: Path to the input image
        style_hint: Optional style guidance
        output_path: Output file path
    """
    model = genai.GenerativeModel('lyria-3-pro')

    with open(image_path, 'rb') as f:
        image_data = base64.b64encode(f.read()).decode('utf-8')

    ext = image_path.split('.')[-1].lower()
    mime_type = {'jpg': 'image/jpeg', 'jpeg': 'image/jpeg', 'png': 'image/png'}.get(ext, 'image/jpeg')

    response = model.generate_content([
        {
            "inline_data": {
                "mime_type": mime_type,
                "data": image_data
            }
        },
        f"Generate music that captures the mood and atmosphere of this image. {style_hint} "
        f"Instrumental only, 2 minutes."
    ])

    for part in response.candidates[0].content.parts:
        if hasattr(part, 'inline_data') and part.inline_data.mime_type.startswith('audio'):
            audio_data = base64.b64decode(part.inline_data.data)
            with open(output_path, 'wb') as f:
                f.write(audio_data)
            return output_path

    raise ValueError("No audio generated")
```

## Creative Use Cases

### Automated Video Background Music

```python
def generate_bgm_for_scenes(scenes: list[dict]) -> list[str]:
    """
    Generate background music for a list of video scenes.

    scenes format:
    [
        {"scene": "Opening", "mood": "hopeful and inspiring", "duration": 30},
        {"scene": "Climax", "mood": "intense and dramatic", "duration": 60},
        {"scene": "Ending", "mood": "peaceful and reflective", "duration": 45},
    ]
    """
    output_files = []

    for i, scene in enumerate(scenes):
        prompt = (
            f"A {scene['mood']} piece of background music, {scene['duration']} seconds long. "
            f"Scene context: {scene['scene']}. Instrumental only. "
            f"Suitable as non-distracting background music for a video."
        )

        output_path = f"bgm_{i:02d}_{scene['scene'].lower().replace(' ', '_')}.mp3"
        generate_music(prompt, output_path)
        output_files.append(output_path)

    return output_files
```

```python
GAME_MUSIC = {
```
