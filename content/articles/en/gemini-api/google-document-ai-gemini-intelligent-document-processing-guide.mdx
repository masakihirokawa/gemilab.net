Processing PDF invoices, contracts, and application forms manually is one of the most persistent sources of human error and operational cost in businesses of every size. Google Document AI and Gemini API together form a powerful combination for intelligent document processing (IDP) that goes well beyond OCR.

Document AI provides specialized processors for structured entity extraction, while Gemini interprets the extracted data for anomaly detection, risk assessment, and decision support. This guide shows you how to build a production-grade IDP pipeline using both services.

## Google Document AI Overview

Document AI offers three main processor categories:

**Pretrained Processors** — Specialized models for common document types: Invoice Parser, Contract Document Analyzer, Form Parser, Identity Document Parser, and more.

**Custom Document Extractor** — Train field-level extraction models on your own document types: medical records, industry-specific forms, proprietary templates.

**Document OCR** — General-purpose OCR when you need text extraction without structured field parsing.

## Invoice Processing Pipeline

### Document AI Invoice Extraction

```python
from google.cloud import documentai_v1 as documentai
from dataclasses import dataclass
from typing import Optional

@dataclass
class InvoiceData:
    invoice_id: str
    supplier_name: str
    invoice_date: Optional[str]
    due_date: Optional[str]
    total_amount: Optional[float]
    line_items: list[dict]
    confidence_score: float

def process_invoice(
    pdf_bytes: bytes,
    processor_name: str,
) -> InvoiceData:
    client = documentai.DocumentProcessorServiceClient()

    raw_document = documentai.RawDocument(
        content=pdf_bytes,
        mime_type="application/pdf",
    )

    result = client.process_document(
        request=documentai.ProcessRequest(
            name=processor_name,
            raw_document=raw_document,
        )
    )

    document = result.document
    entities = {e.type_: e for e in document.entities}

    def get_value(key: str) -> Optional[str]:
        entity = entities.get(key)
        return entity.mention_text if entity else None

    def parse_amount(value: Optional[str]) -> Optional[float]:
        if not value:
            return None
        cleaned = value.replace(",", "").replace("¥", "").replace("$", "").strip()
        try:
            return float(cleaned)
        except ValueError:
            return None

    line_items = []
    for entity in document.entities:
        if entity.type_ == "line_item":
            item = {prop.type_: prop.mention_text for prop in entity.properties}
            line_items.append(item)

    return InvoiceData(
        invoice_id=get_value("invoice_id") or "",
        supplier_name=get_value("supplier_name") or "",
        invoice_date=get_value("invoice_date"),
        due_date=get_value("due_date"),
        total_amount=parse_amount(get_value("total_amount")),
        line_items=line_items,
        confidence_score=min(
            e.confidence for e in document.entities
        ) if document.entities else 0.0,
    )
```

### Gemini API for Anomaly Detection and Analysis

```python
import google.generativeai as genai
import json
from dataclasses import asdict

genai.configure(api_key="YOUR_GEMINI_API_KEY")

def analyze_invoice(
    invoice_data: InvoiceData,
    vendor_history: list[dict],
) -> dict:
    model = genai.GenerativeModel("gemini-2.5-flash")

    prompt = f"""
Analyze this invoice data and assess the following:

## Invoice Data
{json.dumps(asdict(invoice_data), indent=2)}

## Vendor Transaction History (Last 3 Months)
{json.dumps(vendor_history, indent=2)}

## Analysis Required
1. **Anomaly Detection**: Any unusual amounts, items, or quantities?
2. **Consistency Check**: Deviations from historical patterns?
3. **Payment Priority**: Based on due date and business impact (HIGH/MEDIUM/LOW)
4. **Recommended Action**: Any follow-up needed with the vendor?

Respond in JSON:
{{
  "anomalies": [...],
  "consistency_issues": [...],
  "payment_priority": "HIGH|MEDIUM|LOW",
  "priority_reason": "...",
  "recommended_action": "...",
  "confidence": 0.0
}}
"""

    response = model.generate_content(
        prompt,
        generation_config=genai.types.GenerationConfig(temperature=0.1),
    )

    text = response.text.strip().lstrip("```json").lstrip("```").rstrip("```")
    return json.loads(text)
```

## Contract Analysis Pipeline

```python
def analyze_contract(pdf_bytes: bytes, processor_name: str) -> dict:
    client = documentai.DocumentProcessorServiceClient()

    result = client.process_document(
        request=documentai.ProcessRequest(
            name=processor_name,
            raw_document=documentai.RawDocument(
                content=pdf_bytes, mime_type="application/pdf"
            ),
        )
    )

    document = result.document
    contract_data: dict = {
        "parties": [],
        "effective_date": None,
        "expiration_date": None,
        "payment_terms": None,
        "governing_law": None,
    }

    for entity in document.entities:
        if entity.type_ == "contract_party":
            contract_data["parties"].append(entity.mention_text)
        elif entity.type_ in contract_data:
            contract_data[entity.type_] = entity.mention_text

    return contract_data


def assess_contract_risks(contract_text: str, contract_data: dict) -> dict:
    model = genai.GenerativeModel("gemini-2.5-flash")

    prompt = f"""
Analyze this contract for legal risks and critical terms.

## Extracted Metadata
{json.dumps(contract_data, indent=2)}

## Contract Text (excerpt)
{contract_text[:8000]}

## Analysis Required
1. **Risk Clauses**: Unfavorable terms, one-sided provisions, liability waivers
2. **Critical Dates**: Renewal deadlines, notice requirements
3. **Penalty Terms**: Liquidated damages, breach consequences
4. **Termination Conditions**: Exit clauses and procedures
5. **Overall Risk Level**: LOW/MEDIUM/HIGH with justification

Respond in JSON:
{{
  "risk_clauses": [{{"clause": "...", "risk_type": "...", "severity": "LOW|MEDIUM|HIGH"}}],
  "important_dates": [...],
  "penalty_terms": [...],
  "termination_conditions": [...],
  "overall_risk": "LOW|MEDIUM|HIGH",
  "risk_summary": "...",
  "recommended_review_points": [...]
}}
"""

    response = model.generate_content(
        prompt,
        generation_config=genai.types.GenerationConfig(temperature=0.1),
    )
    text = response.text.strip().lstrip("```json").lstrip("```").rstrip("```")
    return json.loads(text)
```

## Scalable Event-Driven Architecture with Cloud Run + Pub/Sub

For processing hundreds or thousands of documents daily, an event-driven architecture handles load spikes gracefully.

```python
# Cloud Run worker: main.py
from flask import Flask, request, jsonify
from google.cloud import storage
import base64, json

app = Flask(__name__)

@app.route('/process', methods=['POST'])
def process_document_event():
    envelope = request.get_json()
    data = base64.b64decode(envelope['message']['data']).decode('utf-8')
    event = json.loads(data)

    pdf_bytes = (
        storage.Client()
        .bucket(event['bucket'])
        .blob(event['name'])
        .download_as_bytes()
    )

    doc_type = event.get('doc_type', 'invoice')

    if doc_type == 'invoice':
        invoice_data = process_invoice(pdf_bytes, INVOICE_PROCESSOR_NAME)
        vendor_history = get_vendor_history(invoice_data.supplier_name)
        analysis = analyze_invoice(invoice_data, vendor_history)
        route = route_by_confidence(invoice_data, analysis)
        save_result(event['name'], invoice_data, analysis, route)

    elif doc_type == 'contract':
        contract_data = analyze_contract(pdf_bytes, CONTRACT_PROCESSOR_NAME)
        contract_text = extract_full_text(pdf_bytes)
        risks = assess_contract_risks(contract_text, contract_data)
        save_contract_result(event['name'], contract_data, risks)

    return jsonify({"status": "processed", "file": event['name']}), 200
```

## Confidence-Based Routing

Not every document needs human review. Route documents automatically based on extraction confidence and anomaly signals.

```python
THRESHOLDS = {
    "auto_approve": 0.95,
    "human_review": 0.80,
    "manual_processing": 0.60,
}

def route_by_confidence(invoice: InvoiceData, analysis: dict) -> str:
    combined = (invoice.confidence_score + analysis.get("confidence", 0)) / 2
    has_anomalies = len(analysis.get("anomalies", [])) > 0
    is_high_priority = analysis.get("payment_priority") == "HIGH"

    if has_anomalies or is_high_priority:
        return "human_review"

    if combined >= THRESHOLDS["auto_approve"]:
        return "auto_approve"
    elif combined >= THRESHOLDS["human_review"]:
        return "human_review"
    else:
        return "manual_processing"
```

## Closing Thoughts
