## Pattern 1: Image-Based Applications

Receipt recognition, product detection, document analysis fall here.

### Example: Ecommerce Auto-Categorization

```python
from google.generativeai import GenerativeModel
import base64

class ProductCategoryClassifier:
    def __init__(self):
        self.model = GenerativeModel('gemini-2.0-flash')
    
    def classify_product(self, image_path):
        """
        Automatically classify product image
        Returns: category, subcategory, confidence score
        """
        
        with open(image_path, 'rb') as f:
            image_data = base64.standard_b64encode(f.read()).decode('utf-8')
        
        prompt = """
        Analyze this product image and provide classification.
        
        Format your response as:
        - Main Category: [category name]
        - Subcategory: [subcategory]
        - Product Type: [specific name]
        - Confidence: [0.0-1.0]
        - Visual Features: [observed characteristics]
        - Recommended Tags: [3-5 SEO tags]
        """
        
        response = self.model.generate_content(
            contents=[
                {"type": "text", "text": prompt},
                {
                    "type": "image",
                    "image": {
                        "mime_type": "image/jpeg",
                        "data": image_data
                    }
                }
            ]
        )
        
        return self._parse_response(response.text)
```

### Performance Optimization

```python
class OptimizedClassifier:
    def classify_batch(self, image_paths, batch_size=10):
        """
        Efficiently classify multiple images with:
        - Batch processing to reduce API calls
        - Caching to prevent redundant computation
        - Parallel execution for faster processing
        """
        
        from functools import lru_cache
        import concurrent.futures
        
        @lru_cache(maxsize=1000)
        def classify_single(image_path):
            return self.classify_product(image_path)
        
        with concurrent.futures.ThreadPoolExecutor(max_workers=5) as executor:
            results = [
                executor.submit(classify_single, path) 
                for path in image_paths
            ]
            return [f.result() for f in results]
```

### Video Processing Optimization

```python
class VideoOptimization:
    @staticmethod
    def optimize_for_api(input_path, output_path):
        """
        Convert to Gemini-optimal format:
        - Codec: H.264
        - Resolution: 720p (1080p acceptable)
        - Frame rate: 30fps
        - Audio: AAC, 128kbps
        - Max size: <100MB
        """
        import subprocess
        
        cmd = [
            'ffmpeg', '-i', input_path,
            '-c:v', 'libx264', '-preset', 'medium',
            '-crf', '23', '-s', '1280x720', '-r', '30',
            '-c:a', 'aac', '-b:a', '128k',
            output_path
        ]
        subprocess.run(cmd, check=True)
    
    @staticmethod
    def extract_keyframes(video_path, max_frames=30):
        """
        Extract only important frames to reduce processing time
        and API cost by 50-70%
        """
        import cv2
        import numpy as np
        
        cap = cv2.VideoCapture(video_path)
        frames, prev_frame = [], None
        
        while True:
            ret, frame = cap.read()
            if not ret:
                break
            
            if prev_frame is not None:
                diff = cv2.absdiff(prev_frame, frame)
                if np.mean(diff) > 50:  # Significant change
                    frames.append(frame)
            
            prev_frame = frame
            if len(frames) >= max_frames:
                break
        
        cap.release()
        return frames
```

### Context Window Optimization

```python
class ContextOptimizer:
    @staticmethod
    def chunk_long_documents(document, chunk_size=4000):
        """
        Split long documents while maintaining context overlap
        Preserves analysis quality across chunks
        """
        words = document.split()
        chunks = []
        overlap = 100
        
        for i in range(0, len(words), chunk_size - overlap):
            chunk = ' '.join(words[i:i+chunk_size])
            chunks.append(chunk)
        
        return chunks
    
    @staticmethod
    def use_prompt_caching():
        """
        Cache expensive system prompts (90% cost reduction on cached tokens)
        
        Avoid resending long system instructions every request
        """
        response = GenerativeModel('gemini-2.0-flash').generate_content(
            contents=[
                {
                    "type": "text",
                    "text": "Long system prompt with rules... (2000 tokens)",
                    "cache_control": True
                },
                {"type": "text", "text": "User query..."}
            ]
        )
        return response
```

## Production Optimization

### Rate Limiting & Retry Logic

```python
class RobustAPIClient:
    def __init__(self, max_retries=3, initial_backoff=1):
        self.model = GenerativeModel('gemini-2.0-flash')
        self.max_retries = max_retries
        self.initial_backoff = initial_backoff
    
    def call_with_retry(self, contents):
        """
        Handle rate limiting (429 errors) with exponential backoff
        """
        import time
        import random
        
        for attempt in range(self.max_retries):
            try:
                return self.model.generate_content(contents=contents)
            except Exception as e:
                if "429" in str(e):
                    wait = self.initial_backoff * (2 ** attempt) + random.uniform(0, 1)
                    print(f"Rate limited. Retrying in {wait:.1f}s...")
                    time.sleep(wait)
                else:
                    raise
        
        raise Exception(f"Failed after {self.max_retries} retries")
```

### Cost Monitoring

```python
class CostMonitor:
    PRICING = {
        "text_input": 0.075 / 1_000_000,     # $/token
        "text_output": 0.3 / 1_000_000,
        "image": 0.001,                       # $/image
        "video": 0.001,                       # $/second
        "audio": 0.0001                       # $/second
    }
    
    @staticmethod
    def estimate_cost(contents):
        """
        Estimate cost before sending request
        """
        cost = 0
        
        for item in contents:
            if item["type"] == "text":
                tokens = len(item["text"]) / 4  # ~1 token per 4 chars
                cost += tokens * CostMonitor.PRICING["text_input"]
            elif item["type"] == "image":
                cost += CostMonitor.PRICING["image"]
            elif item["type"] == "video":
                cost += 120 * CostMonitor.PRICING["video"]  # Assume 2 min
        
        return cost
```
