LINE is the dominant messaging platform in Japan and across much of East and Southeast Asia, with over 97 million monthly active users in Japan alone. Building an AI-powered chatbot on LINE gives you direct access to this massive user base — whether for customer support automation, personal assistants, or interactive marketing tools.

In this tutorial, you'll learn how to combine Google's Gemini API with the LINE Messaging API to build a Python-based chatbot that understands context, handles multi-turn conversations, and even analyzes images users send.

- Creating a LINE Messaging API channel on LINE Developers
- Setting up the Gemini API with the Google AI Python SDK
- Building a Flask webhook server that routes messages through Gemini
- Implementing multi-turn conversation history
- Deploying to Google Cloud Run for production use

## Building the LINE Bot Server

### Main Application

Here's the complete Flask webhook server that receives LINE messages and generates responses with the Gemini API:

```python
# app.py
import os
from flask import Flask, request, abort
from google import genai
from linebot.v3 import WebhookParser
from linebot.v3.messaging import (
    Configuration,
    ApiClient,
    MessagingApi,
    ReplyMessageRequest,
    TextMessage,
)
from linebot.v3.webhooks import MessageEvent, TextMessageContent

app = Flask(__name__)

# --- Configuration ---
GEMINI_API_KEY = os.environ["GEMINI_API_KEY"]
LINE_CHANNEL_SECRET = os.environ["LINE_CHANNEL_SECRET"]
LINE_CHANNEL_ACCESS_TOKEN = os.environ["LINE_CHANNEL_ACCESS_TOKEN"]

# Initialize Gemini client
gemini_client = genai.Client(api_key=GEMINI_API_KEY)

# Initialize LINE API
configuration = Configuration(access_token=LINE_CHANNEL_ACCESS_TOKEN)
parser = WebhookParser(LINE_CHANNEL_SECRET)

# Per-user conversation history (in-memory)
conversation_history: dict[str, list[dict]] = {}
MAX_HISTORY = 20  # Keep the most recent 20 messages

def get_gemini_response(user_id: str, user_message: str) -> str:
    """Generate a multi-turn response using the Gemini API."""
    # Retrieve or initialize conversation history
    if user_id not in conversation_history:
        conversation_history[user_id] = []

    history = conversation_history[user_id]

    # Append the user message
    history.append({"role": "user", "parts": [{"text": user_message}]})

    # Call the Gemini API
    response = gemini_client.models.generate_content(
        model="gemini-3-flash",
        contents=history,
        config={
            "system_instruction": (
                "You are a friendly AI assistant running on LINE. "
                "Keep your answers concise (under 200 words) and "
                "conversational. Be helpful and approachable."
            ),
            "temperature": 0.7,
            "max_output_tokens": 500,
        },
    )

    assistant_message = response.text

    # Append the assistant response to history
    history.append({"role": "model", "parts": [{"text": assistant_message}]})

    # Trim history if it gets too long
    if len(history) > MAX_HISTORY:
        conversation_history[user_id] = history[-MAX_HISTORY:]

    return assistant_message

@app.route("/callback", methods=["POST"])
def callback():
    """LINE Webhook endpoint."""
    signature = request.headers.get("X-Line-Signature", "")
    body = request.get_data(as_text=True)

    try:
        events = parser.parse(body, signature)
    except Exception:
        abort(400)

    for event in events:
        if isinstance(event, MessageEvent) and isinstance(
            event.message, TextMessageContent
        ):
            user_id = event.source.user_id
            user_message = event.message.text

            # Generate a response with Gemini
            reply_text = get_gemini_response(user_id, user_message)

            # Send the reply via LINE
            with ApiClient(configuration) as api_client:
                line_bot_api = MessagingApi(api_client)
                line_bot_api.reply_message(
                    ReplyMessageRequest(
                        reply_token=event.reply_token,
                        messages=[TextMessage(text=reply_text)],
                    )
                )

    return "OK"

@app.route("/health", methods=["GET"])
def health():
    """Health check endpoint."""
    return "OK"

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=8080, debug=True)
```

### Key Design Decisions

- **`conversation_history`**: A dictionary keyed by user ID that stores the conversation context. This enables natural follow-up messages like "tell me more about that" without losing context
- **`system_instruction`**: Tells Gemini how to behave as a LINE bot — controlling tone, response length, and personality
- **`MAX_HISTORY`**: Caps memory usage at 20 messages per user. For production, consider persisting history in Redis or Firestore instead of in-memory storage

## Building a Basic Text Response Bot

Let's start with a simple bot that responds to user messages using the Gemini API.

```python
import os
import logging
from dotenv import load_dotenv
from google import genai
from telegram import Update
from telegram.ext import (
    ApplicationBuilder,
    CommandHandler,
    MessageHandler,
    ContextTypes,
    filters,
)

# Load environment variables
load_dotenv()

# Configure logging
logging.basicConfig(
    format="%(asctime)s - %(name)s - %(levelname)s - %(message)s",
    level=logging.INFO,
)
logger = logging.getLogger(__name__)

# Initialize the Gemini client
client = genai.Client(api_key=os.getenv("GEMINI_API_KEY"))

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """
    Handler for the /start command
    """
    await update.message.reply_text(
        "Hello! I'm a Gemini AI assistant.\n"
        "Send me text or images and I'll analyze and respond.\n\n"
        "Commands:\n"
        "/start - About this bot\n"
        "/reset - Clear conversation history"
    )

async def reset(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """
    Clear conversation history with /reset
    """
    context.user_data.pop("history", None)
    await update.message.reply_text("Conversation history cleared.")

async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """
    Respond to text messages using the Gemini API
    """
    user_message = update.message.text

    # Manage conversation history (max 20 turns)
    history = context.user_data.get("history", [])
    history.append({"role": "user", "parts": [{"text": user_message}]})

    # Trim history to prevent excessive token usage
    if len(history) > 40:
        history = history[-40:]

    try:
        # Call the Gemini API
        response = client.models.generate_content(
            model="gemini-3-flash",
            contents=history,
            config={
                "system_instruction": (
                    "You are a helpful and knowledgeable AI assistant. "
                    "Respond in the same language the user writes in. "
                    "Keep your answers concise and accurate."
                ),
                "max_output_tokens": 2048,
                "temperature": 0.7,
            },
        )

        reply = response.text

        # Add assistant response to history
        history.append({"role": "model", "parts": [{"text": reply}]})
        context.user_data["history"] = history

        # Handle Telegram's 4096 character message limit
        if len(reply) > 4000:
            for i in range(0, len(reply), 4000):
                await update.message.reply_text(reply[i : i + 4000])
        else:
            await update.message.reply_text(reply)

    except Exception as e:
        logger.error(f"Gemini API error: {e}")
        await update.message.reply_text(
            "Sorry, an error occurred. Please try again in a moment."
        )

def main():
    """
    Launch the bot
    """
    app = (
        ApplicationBuilder()
        .token(os.getenv("TELEGRAM_BOT_TOKEN"))
        .build()
    )

    app.add_handler(CommandHandler("start", start))
    app.add_handler(CommandHandler("reset", reset))
    app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, handle_message))

    logger.info("Bot is running...")
    app.run_polling()

if __name__ == "__main__":
    main()
```

Once running, simply send a message to your bot on Telegram and it will respond using the Gemini API. Conversation history is maintained automatically, enabling contextual multi-turn conversations.

## Going Multimodal — Adding Image Analysis

One of the Gemini API's greatest strengths is its multimodal capability. Let's add image analysis so users can send photos for the AI to interpret.

```python
import io
import httpx

async def handle_photo(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """
    Receive and analyze images with Gemini
    """
    # Get the highest resolution version of the photo
    photo = update.message.photo[-1]
    file = await context.bot.get_file(photo.file_id)

    # Download the image data
    async with httpx.AsyncClient() as http_client:
        response = await http_client.get(file.file_path)
        image_bytes = response.content

    # Use the caption as the user's question, or fall back to a default
    caption = update.message.caption or "Please describe this image in detail."

    try:
        await update.message.reply_text("Analyzing your image...")

        # Send multimodal request to the Gemini API
        response = client.models.generate_content(
            model="gemini-3-flash",
            contents=[
                {
                    "role": "user",
                    "parts": [
                        {"text": caption},
                        {
                            "inline_data": {
                                "mime_type": "image/jpeg",
                                "data": __import__("base64").b64encode(
                                    image_bytes
                                ).decode(),
                            }
                        },
                    ],
                }
            ],
            config={
                "max_output_tokens": 2048,
                "temperature": 0.5,
            },
        )

        await update.message.reply_text(response.text)

    except Exception as e:
        logger.error(f"Image analysis error: {e}")
        await update.message.reply_text(
            "An error occurred while analyzing the image. Please try a different one."
        )
```

Add the following handler to your `main()` function:

```python
app.add_handler(MessageHandler(filters.PHOTO, handle_photo))
```

Now when users send a photo, Gemini will analyze its contents and respond. They can add a caption like "How many calories are in this dish?" or "How do I fix this error?" for more targeted analysis.

## Adding Document Analysis

You can also let users send PDFs and text files for AI-powered analysis.

```python
async def handle_document(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """
    Receive and analyze documents (PDF, text) with Gemini
    """
    document = update.message.document

    # Supported MIME types
    supported_types = {
        "application/pdf": "application/pdf",
        "text/plain": "text/plain",
        "text/csv": "text/csv",
    }

    mime_type = document.mime_type
    if mime_type not in supported_types:
        await update.message.reply_text(
            "Unsupported file format. Please send a PDF, text, or CSV file."
        )
        return

    # File size check (20MB limit)
    if document.file_size > 20 * 1024 * 1024:
        await update.message.reply_text(
            "File is too large (limit: 20MB)."
        )
        return

    file = await context.bot.get_file(document.file_id)

    async with httpx.AsyncClient() as http_client:
        response = await http_client.get(file.file_path)
        file_bytes = response.content

    caption = update.message.caption or "Please summarize this document."

    try:
        await update.message.reply_text("Analyzing your document...")

        import base64
        response = client.models.generate_content(
            model="gemini-3-flash",
            contents=[
                {
                    "role": "user",
                    "parts": [
                        {"text": caption},
                        {
                            "inline_data": {
                                "mime_type": mime_type,
                                "data": base64.b64encode(file_bytes).decode(),
                            }
                        },
                    ],
                }
            ],
            config={
                "max_output_tokens": 4096,
                "temperature": 0.3,
            },
        )

        reply = response.text
        if len(reply) > 4000:
            for i in range(0, len(reply), 4000):
                await update.message.reply_text(reply[i : i + 4000])
        else:
            await update.message.reply_text(reply)

    except Exception as e:
        logger.error(f"Document analysis error: {e}")
        await update.message.reply_text(
            "An error occurred while analyzing the document."
        )
```

## Error Handling and Rate Limiting

For production deployments, robust error handling is essential. Here's a retry wrapper with exponential backoff for handling API rate limits and transient failures.

```python
import asyncio
from google.api_core.exceptions import ResourceExhausted, ServiceUnavailable

async def call_gemini_with_retry(contents, config, max_retries=3):
    """
    Call the Gemini API with automatic retry on rate limits
    """
    for attempt in range(max_retries):
        try:
            response = client.models.generate_content(
                model="gemini-3-flash",
                contents=contents,
                config=config,
            )
            return response

        except ResourceExhausted:
            # Rate limited — exponential backoff
            wait_time = 2 ** attempt
            logger.warning(
                f"Rate limited. Retrying in {wait_time}s "
                f"(attempt {attempt + 1}/{max_retries})"
            )
            await asyncio.sleep(wait_time)

        except ServiceUnavailable:
            # Service temporarily down — retry
            wait_time = 3 ** attempt
            logger.warning(
                f"Service unavailable. Retrying in {wait_time}s "
                f"(attempt {attempt + 1}/{max_retries})"
            )
            await asyncio.sleep(wait_time)

    raise Exception("Maximum retries exceeded")

# response = await call_gemini_with_retry(
#     contents=history,
#     config={"max_output_tokens": 2048, "temperature": 0.7},
# )
```
