## What Is Function Calling? Connecting AI to the Real World

Gemini API's **Function Calling** (also referred to as tool use) is the feature that allows AI models to interact with external functions and APIs. While a standard LLM only generates text, Function Calling lets your application trigger real actions — checking the weather, querying a database, scheduling a calendar event, and much more.

As of 2026, Function Calling is supported across all major Gemini models, including Gemini 2.5 Pro, Gemini 2.5 Flash, and the Gemini 3 series. It's widely used by individual developers and enterprise teams alike.

Here's how the flow works at a high level:

1. You define "available functions" and pass them to the API alongside the user's message
2. The user asks something like "What's the weather in Tokyo?"
3. Gemini decides that `get_weather(city="Tokyo")` is the right call and returns that function invocation
4. Your application actually runs the function and sends the result back to the model
5. Gemini generates a natural-language response based on the result

This division of labor — AI decides, app executes — is the essence of Function Calling.

## Your First Function Calling Implementation

Let's walk through a weather-retrieval example using dummy data — no real weather API needed to understand the mechanics.

### Step 1: Define the Function (Tool Declaration)

```python
import json
from google import genai
from google.genai import types

# The actual Python function your app will call (dummy implementation)
def get_weather(city: str) -> dict:
    """Returns weather info for a given city (dummy data)."""
    weather_data = {
        "Tokyo": {"condition": "Sunny", "temperature": 22, "humidity": 55},
        "Osaka": {"condition": "Cloudy", "temperature": 19, "humidity": 68},
        "Sapporo": {"condition": "Snowy", "temperature": -3, "humidity": 85},
    }
    return weather_data.get(city, {"condition": "Unknown", "temperature": 0, "humidity": 0})

# The tool definition passed to Gemini
weather_tool = types.Tool(
    function_declarations=[
        types.FunctionDeclaration(
            name="get_weather",
            description="Retrieves current weather info (condition, temperature, humidity) for a given city",
            parameters=types.Schema(
                type=types.Type.OBJECT,
                properties={
                    "city": types.Schema(
                        type=types.Type.STRING,
                        description="The city name to check weather for (e.g., Tokyo, Osaka)"
                    )
                },
                required=["city"]
            )
        )
    ]
)
```

### Step 2: Send the Request and Receive the Function Call

```python
client = genai.Client(api_key=os.environ["YOUR_GEMINI_API_KEY"])

# Send the user's message along with the tool definition
response = client.models.generate_content(
    model="gemini-2.5-flash",
    contents="What's the weather like in Tokyo today?",
    config=types.GenerateContentConfig(
        tools=[weather_tool]
    )
)

# Check if Gemini responded with a function call
for part in response.candidates[0].content.parts:
    if part.function_call:
        func_call = part.function_call
        print(f"Function: {func_call.name}")
        print(f"Args: {dict(func_call.args)}")
        # Expected output:
        # Function: get_weather
        # Args: {'city': 'Tokyo'}
```

### Step 3: Execute the Function and Return the Result

```python
# Retrieve the function call from the response
func_call = response.candidates[0].content.parts[0].function_call

# Execute the actual function
if func_call.name == "get_weather":
    result = get_weather(**dict(func_call.args))
    print(f"Function result: {result}")
    # Expected: {'condition': 'Sunny', 'temperature': 22, 'humidity': 55}

# Send the result back to Gemini to generate the final answer
final_response = client.models.generate_content(
    model="gemini-2.5-flash",
    contents=[
        types.Content(role="user", parts=[
            types.Part(text="What's the weather like in Tokyo today?")
        ]),
        response.candidates[0].content,  # The model's function call
        types.Content(role="tool", parts=[
            types.Part(
                function_response=types.FunctionResponse(
                    name="get_weather",
                    response=result
                )
            )
        ])
    ],
    config=types.GenerateContentConfig(tools=[weather_tool])
)

print(final_response.text)
# Expected: "It's sunny in Tokyo today with a temperature of 22°C and humidity at 55%. A pleasant day!"
```

Once you understand this three-step loop, the rest of Function Calling is just variations on this theme.

## Common Errors and How to Fix Them

### Error 1: Function Return Value Is Not JSON-Serializable

```python
# ❌ Problem: datetime objects can't be JSON-serialized
def get_schedule(date: str) -> dict:
    from datetime import datetime
    return {"event": "Meeting", "time": datetime.now()}  # datetime → fails

# ✅ Fix: Convert to a string
def get_schedule(date: str) -> dict:
    from datetime import datetime
    return {"event": "Meeting", "time": datetime.now().isoformat()}  # ISO string → works
```

### Error 2: Model Returns Plain Text Instead of a Function Call

```python
# ❌ Vague description — Gemini can't decide when to use this
types.FunctionDeclaration(
    name="get_data",
    description="Gets data",  # Too ambiguous
    ...
)

# ✅ Specific description — clear trigger conditions
types.FunctionDeclaration(
    name="get_weather",
    description="Retrieves real-time weather data (condition, temperature, humidity, wind speed) for a specified city",
    ...
)
```

### Error 3: `finish_reason` Is `MAX_TOKENS` Instead of `STOP`

Long Function Calling conversations can exceed the context window. In this case, switch to `gemini-2.5-pro` (2M token context) or implement conversation history pruning.

## How Function Calling Actually Works

Before diving into implementation, it's worth being precise about the flow:

1. Your app sends Gemini a set of function definitions alongside the user's message
2. If Gemini decides a function call is needed, it returns a `functionCall` response (not a text answer)
3. Your app executes the actual function and sends the result back to Gemini
4. Gemini uses the result to generate a final response

This exchange can repeat multiple times in a single turn — multi-turn Function Calling. Gemini might receive the first function's result and decide it needs another function call before it can answer.

## Function Schema Design: Where Accuracy Is Won or Lost

The quality of your function definitions determines how accurately Gemini uses them. Gemini reads your descriptions to decide when to call a function, which function to call, and what arguments to pass.

**Vague definition** — Gemini makes poor decisions:
```javascript
{
  name: "search",
  description: "Searches for something",
  parameters: {
    type: "object",
    properties: {
      query: { type: "string" }
    }
  }
}
```

**Specific definition** — Gemini makes good decisions:
```javascript
{
  name: "search_products",
  description: "Search the product database by name, category, or price range. Use when the user is looking for specific products or asking for a product listing. For inventory availability, use check_inventory instead.",
  parameters: {
    type: "object",
    properties: {
      query: {
        type: "string",
        description: "Search keyword (product name, category, etc.)"
      },
      max_price: {
        type: "number",
        description: "Maximum price in USD. Omit for no upper limit."
      },
      in_stock_only: {
        type: "boolean",
        description: "If true, return only in-stock products",
        default: false
      }
    },
    required: ["query"]
  }
}
```

The most important parts of the description: when to use the function and how it differs from similar functions. When multiple functions have overlapping scope, Gemini needs explicit boundaries to choose correctly.

## TypeScript Implementation

```typescript
import { GoogleGenerativeAI, FunctionDeclarationSchemaType } from "@google/generative-ai";

const genAI = new GoogleGenerativeAI("YOUR_GEMINI_API_KEY");

const tools = [
  {
    functionDeclarations: [
      {
        name: "get_weather",
        description:
          "Get current weather for a specified city. Returns temperature, conditions, and humidity. Use when the user asks about weather.",
        parameters: {
          type: FunctionDeclarationSchemaType.OBJECT,
          properties: {
            city: {
              type: FunctionDeclarationSchemaType.STRING,
              description: "City name to check weather for (e.g. Tokyo, London)",
            },
            unit: {
              type: FunctionDeclarationSchemaType.STRING,
              description: "Temperature unit: 'celsius' or 'fahrenheit'",
              enum: ["celsius", "fahrenheit"],
            },
          },
          required: ["city"],
        },
      },
    ],
  },
];

const functions: Record<string, (args: Record<string, unknown>) => Promise<unknown>> = {
  get_weather: async ({ city, unit = "celsius" }) => {
    const response = await fetchWeatherAPI(city as string, unit as string);
    return {
      city,
      temperature: response.temp,
      condition: response.weather,
      humidity: response.humidity,
    };
  },
};

async function chat(userMessage: string): Promise<string> {
  const model = genAI.getGenerativeModel({ model: "gemini-2.5-pro", tools });
  const chatSession = model.startChat({ history: [] });

  let response = await chatSession.sendMessage(userMessage);

  // Function Calling loop — capped at 10 iterations
  let loopCount = 0;
  while (loopCount < 10) {
    const candidate = response.response.candidates?.[0];
    if (\!candidate) break;

    const functionCallParts = candidate.content.parts.filter(
      (part) => "functionCall" in part
    );

    if (functionCallParts.length === 0) {
      return response.response.text();
    }

    const functionResponses = await Promise.all(
      functionCallParts.map(async (part) => {
        if (\!("functionCall" in part)) return null;
        const { name, args } = part.functionCall;

        if (\!functions[name]) {
          return {
            functionResponse: {
              name,
              response: { error: `Function '${name}' not found` },
            },
          };
        }

        try {
          const result = await functions[name](args as Record<string, unknown>);
          return { functionResponse: { name, response: result } };
        } catch (error) {
          return {
            functionResponse: {
              name,
              response: { error: `Execution failed: ${String(error)}` },
            },
          };
        }
      })
    );

    response = await chatSession.sendMessage(
      functionResponses.filter(Boolean) as Parameters<typeof chatSession.sendMessage>[0]
    );
    loopCount++;
  }

  return "The request became too complex to resolve. Try a more specific question.";
}
```

The loop cap is essential. One of the most expensive bugs in Function Calling is an infinite loop — Gemini repeatedly calling the same function and never reaching a text response. Without a limit, API costs can spike before you notice.

## Parallel Function Calling

Gemini 2.5 can request multiple functions simultaneously in a single response:

```
User: What's the weather in Tokyo and London?

Gemini responds:
[
  { functionCall: { name: "get_weather", args: { city: "Tokyo" } } },
  { functionCall: { name: "get_weather", args: { city: "London" } } }
]
```

The `Promise.all()` in the implementation above handles this automatically — both weather calls execute in parallel rather than sequentially.

For parallel calling to work well, functions should represent independent operations. Functions with dependencies (where B needs A's output) won't benefit from parallel execution.

## Error Handling

Function Calling errors fall into two categories that need different handling.

**Function execution errors** — the function was called but failed. Return the error as a response rather than throwing:

```typescript
try {
  const result = await functions[name](args);
  return { functionResponse: { name, response: result } };
} catch (error) {
  // Don't throw — send the error back to Gemini
  return {
    functionResponse: {
      name,
      response: {
        error: "Failed to retrieve data",
        reason: String(error),
        suggestion: "Please try again in a moment",
      },
    },
  };
}
```

Returning the error as a response lets Gemini understand the situation and generate an appropriate answer — "I wasn't able to get that information right now, but..." — rather than crashing.

**API errors** — the Gemini API call itself fails. Use exponential backoff for rate limits:

```typescript
async function sendMessageWithRetry(
  chatSession: ReturnType<typeof model.startChat>,
  message: Parameters<typeof chatSession.sendMessage>[0],
  maxRetries = 3
) {
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      return await chatSession.sendMessage(message);
    } catch (error) {
      if (attempt === maxRetries) throw error;

      const isRateLimit = String(error).includes("429") || String(error).includes("RESOURCE_EXHAUSTED");
      const delay = isRateLimit ? 1000 * Math.pow(2, attempt) : 500 * attempt;
      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }
  throw new Error("Max retries exceeded");
}
```

## Debugging Function Calling Issues

**Gemini isn't calling the function**

The most common issue: a question that should trigger a function call gets a text-only response instead.

The usual cause is an ambiguous description that doesn't clearly state when the function should be used. Gemini defaults to generating text when it's uncertain.

Diagnose by forcing function calls:

```typescript
const model = genAI.getGenerativeModel({
  model: "gemini-2.5-pro",
  tools,
  toolConfig: {
    functionCallingConfig: { mode: "ANY" }, // force a function call
  },
});
```

If `ANY` mode works, the problem is in the description or Gemini's `AUTO` mode judgment. Refine the description or use `ANY` mode for specific use cases that always require a function.

**Over-calling**

Functions being called in situations where they shouldn't be. Add explicit negative conditions to the description:

```javascript
description: "Use only when the user is asking about specific products. Do not use for general questions or conversational messages."
```

**Debugging loop behavior**

Log each iteration's function calls to understand how Gemini is reasoning:

```typescript
let loopCount = 0;
while (loopCount < 10) {
  // ... (loop logic)
  console.log(`Loop ${loopCount}: functions called = ${functionCallParts.map(p => p.functionCall?.name).join(", ")}`);
  loopCount++;
}
```

## The Three Things That Matter

After building several production Function Calling systems, I keep coming back to the same lessons:

1. **Write descriptions that say when NOT to use the function.** Gemini's accuracy with function selection is mostly determined by the quality of the description.

2. **Always cap the loop.** Infinite Function Calling loops happen in production. The cap isn't optional.

3. **Return errors as responses, not exceptions.** Gemini handles error responses gracefully. Exceptions just crash the loop.

These three things are the difference between a system that "mostly works" and one that holds up reliably.

## Setup and context — Extending Gemini's Capabilities with Function Calling

Gemini excels at text generation, but it can't directly access real-time weather data, live database contents, or other information outside its training data. Function Calling breaks through this limitation.

With Function Calling, Gemini decides when to invoke an external tool and automatically generates the required parameters. You execute the actual API call with those parameters and return the results to Gemini, which then incorporates the live data into a natural response.

## How Function Calling Works

Function Calling operates in three steps.

1. **Define functions**: Tell Gemini about available functions — their names, descriptions, and parameters
2. **Gemini decides**: Gemini analyzes the user's question and selects the right function with appropriate parameters
3. **Integrate results**: Return function execution results to Gemini for natural language response generation

```python
# Function Calling flow (conceptual)
# User: "What's the weather in Tokyo?"
#   ↓
# Gemini: Decides to call get_weather(city="Tokyo")
#   ↓
# Your code: Actually calls the weather API
#   ↓
# Gemini: "It's currently sunny in Tokyo at 22°C"
```

## Basic Implementation: Weather Lookup

Let's start with the simplest example — fetching weather information.

```python
import google.generativeai as genai

# Configure API key
genai.configure(api_key="YOUR_API_KEY")

# Step 1: Define the function
get_weather = genai.protos.FunctionDeclaration(
    name="get_weather",
    description="Get current weather information for a specified city",
    parameters=genai.protos.Schema(
        type=genai.protos.Type.OBJECT,
        properties={
            "city": genai.protos.Schema(
                type=genai.protos.Type.STRING,
                description="City name (e.g., Tokyo, New York)",
            ),
            "unit": genai.protos.Schema(
                type=genai.protos.Type.STRING,
                enum=["celsius", "fahrenheit"],
                description="Temperature unit (default: celsius)",
            ),
        },
        required=["city"],
    ),
)

# Pass tool definition to the model
tool = genai.protos.Tool(function_declarations=[get_weather])
model = genai.GenerativeModel(
    model_name="gemini-2.5-pro",
    tools=[tool],
)

# Step 2: Send user's question
chat = model.start_chat()
response = chat.send_message("What's the weather like in Tokyo?")

# When Gemini requests a function call
function_call = response.candidates[0].content.parts[0].function_call
print(f"Function: {function_call.name}")
print(f"Parameters: {dict(function_call.args)}")

# Expected output:
# Function: get_weather
# Parameters: {'city': 'Tokyo', 'unit': 'celsius'}
```

### Executing the API Call and Returning Results

```python
import requests

def call_weather_api(city: str, unit: str = "celsius") -> dict:
    """Call the actual weather API."""
    api_key = "YOUR_WEATHER_API_KEY"
    url = "https://api.openweathermap.org/data/2.5/weather"
    params = {
        "q": city,
        "appid": api_key,
        "units": "metric" if unit == "celsius" else "imperial",
    }
    resp = requests.get(url, params=params)
    data = resp.json()
    return {
        "city": city,
        "temperature": data["main"]["temp"],
        "condition": data["weather"][0]["description"],
        "humidity": data["main"]["humidity"],
    }

# Step 3: Execute the function and return results to Gemini
weather_data = call_weather_api(
    city=function_call.args["city"],
    unit=function_call.args.get("unit", "celsius"),
)

# Send results back to Gemini
response = chat.send_message(
    genai.protos.Content(
        parts=[
            genai.protos.Part(
                function_response=genai.protos.FunctionResponse(
                    name="get_weather",
                    response={"result": weather_data},
                )
            )
        ]
    )
)

print(response.text)
# Expected output:
# The current weather in Tokyo is sunny with a temperature of 22°C
# and 45% humidity. It's a comfortable day!
```

## Error Handling and Best Practices

### Validating Function Calls

```python
def handle_function_call(response) -> dict | None:
    """Safely process a Function Call response."""
    try:
        candidate = response.candidates[0]
        part = candidate.content.parts[0]

        # Check if a function call is present
        if not hasattr(part, "function_call") or not part.function_call.name:
            return None

        fc = part.function_call
        return {
            "name": fc.name,
            "args": dict(fc.args),
        }
    except (IndexError, AttributeError) as e:
        print(f"Failed to parse Function Call: {e}")
        return None

# Usage
result = handle_function_call(response)
if result:
    print(f"Execute {result['name']} with {result['args']}")
else:
    print(f"Text response: {response.text}")
```

### Best Practices

Here are the key principles for effective Function Calling.

**Write specific function descriptions**: Instead of "get data," write "search the product database by keyword and return product name, price, and stock count." Vague descriptions lead to Gemini making poor function selection decisions.

**Specify types and defaults for parameters**: Keep `required` parameters to a minimum, and include default values in optional parameter descriptions. This gives Gemini more flexibility in how it calls your functions.

**Return meaningful error messages**: When a function execution fails, return the error reason rather than an empty result. Gemini understands error messages and can provide users with appropriate explanations.

## Setup and context: Function Calling + Structured Output Together

Gemini's **Function Calling** and **responseSchema (structured output)** are distinct features — but together they form the backbone of reliable, type-safe AI pipelines.

```
Function Calling:      LLM invokes external tools/APIs
       ↕
responseSchema:        LLM output is constrained to a JSON schema
```

```python
import json

genai.configure(api_key="YOUR_API_KEY")

WEATHER_TOOL = genai.protos.Tool(
    function_declarations=[
        genai.protos.FunctionDeclaration(

            description="Get current weather for a specified city",

                properties={
                    "city": genai.protos.Schema(
                        type=genai.protos.Type.STRING,
                        description="City name (e.g. Tokyo, New York)",
                    ),
                    "unit": genai.protos.Schema(
                        type=genai.protos.Type.STRING,
                        enum=["celsius", "fahrenheit"],
                    ),
                },
                required=["city"],
            ),
        )
    ]

def get_weather(city: str, unit: str = "celsius") -> dict:
    # In production: call Open-Meteo or OpenWeatherMap
    mock = {
        "Tokyo": {"temp": 22, "condition": "Clear", "humidity": 60},
        "New York": {"temp": 18, "condition": "Partly Cloudy", "humidity": 55},
    }
    data = mock.get(city, {"temp": 20, "condition": "Unknown", "humidity": 50})
    temp = data["temp"]
    if unit == "fahrenheit":
        temp = temp * 9 / 5 + 32
    return {**data, "temp": temp, "unit": unit, "city": city}

TOOL_REGISTRY = {"get_weather": get_weather}

def dispatch_tool_call(function_call) -> str:
    func = TOOL_REGISTRY.get(function_call.name)
    if not func:
        return json.dumps({"error": f"Unknown tool: {function_call.name}"})
    try:
        return json.dumps(func(**dict(function_call.args)))
    except Exception as e:
        return json.dumps({"error": str(e)})

def run_with_tools(user_message: str) -> str:
    model = genai.GenerativeModel(
        model_name="gemini-2.0-flash",
        tools=[WEATHER_TOOL],
    )
    chat = model.start_chat()
    response = chat.send_message(user_message)

    while True:
        parts = response.candidates[0].content.parts
        function_calls = [p for p in parts if hasattr(p, "function_call")]
        if not function_calls:
            break

        tool_responses = []
        for part in function_calls:
            result = dispatch_tool_call(part.function_call)
            tool_responses.append(
                genai.protos.Part(
                    function_response=genai.protos.FunctionResponse(
                        name=part.function_call.name,
                        response={"result": json.loads(result)},
                    )
                )
            )
        response = chat.send_message(tool_responses)

    return response.text

if __name__ == "__main__":
    print(run_with_tools("Compare the weather in Tokyo and New York today."))
```

## 3. Parallel Function Calling

```python
# parallel_function_calling.py
import asyncio
import json
import google.generativeai as genai

class ParallelToolAgent:
    """Agent supporting parallel tool execution for reduced latency"""

    def __init__(self, tools: list, model_name: str = "gemini-2.0-flash"):
        self.model = genai.GenerativeModel(
            model_name=model_name,
            tools=tools,
            tool_config={"function_calling_config": {"mode": "AUTO"}},
        )
        self.tool_registry: dict = {}

    def register_tool(self, name: str, func):
        self.tool_registry[name] = func

    async def _call_tool_async(self, function_call) -> genai.protos.Part:
        func = self.tool_registry.get(function_call.name)
        if not func:
            result = {"error": f"Unknown tool: {function_call.name}"}
        else:
            try:
                args = dict(function_call.args)
                if asyncio.iscoroutinefunction(func):
                    result = await func(**args)
                else:
                    loop = asyncio.get_event_loop()
                    result = await loop.run_in_executor(None, lambda: func(**args))
            except Exception as e:
                result = {"error": str(e)}

        return genai.protos.Part(
            function_response=genai.protos.FunctionResponse(
                name=function_call.name,
                response={"result": result},
            )
        )

    async def run(self, user_message: str, max_iterations: int = 10) -> str:
        loop = asyncio.get_event_loop()
        chat = self.model.start_chat()
        response = await loop.run_in_executor(
            None, lambda: chat.send_message(user_message)
        )

        for _ in range(max_iterations):
            parts = response.candidates[0].content.parts
            calls = [p for p in parts if hasattr(p, "function_call") and p.function_call.name]
            if not calls:
                break

            # Execute ALL tool calls in parallel — key performance optimization
            results = await asyncio.gather(
                *[self._call_tool_async(p.function_call) for p in calls]
            )
            response = await loop.run_in_executor(
                None, lambda: chat.send_message(list(results))
            )

        return response.text

# Example: Stock + News + FX in parallel
async def get_stock(symbol: str) -> dict:
    await asyncio.sleep(0.5)  # Simulate API latency
    return {"symbol": symbol, "price": 875.20, "change": "+1.8%"}

async def get_news(topic: str) -> dict:
    await asyncio.sleep(0.3)
    return {"topic": topic, "headlines": [f"{topic} news 1", f"{topic} news 2"]}

async def get_fx(from_cur: str, to_cur: str) -> dict:
    await asyncio.sleep(0.2)
    return {"pair": f"{from_cur}/{to_cur}", "rate": 149.5}

async def main():
    TOOLS = [
        genai.protos.Tool(function_declarations=[
            genai.protos.FunctionDeclaration(
                name="get_stock",
                description="Get stock price",
                parameters=genai.protos.Schema(
                    type=genai.protos.Type.OBJECT,
                    properties={"symbol": genai.protos.Schema(type=genai.protos.Type.STRING)},
                    required=["symbol"],
                ),
            ),
            genai.protos.FunctionDeclaration(
                name="get_news",
                description="Get latest news",
                parameters=genai.protos.Schema(
                    type=genai.protos.Type.OBJECT,
                    properties={"topic": genai.protos.Schema(type=genai.protos.Type.STRING)},
                    required=["topic"],
                ),
            ),
            genai.protos.FunctionDeclaration(
                name="get_fx",
                description="Get exchange rate",
                parameters=genai.protos.Schema(
                    type=genai.protos.Type.OBJECT,
                    properties={
                        "from_cur": genai.protos.Schema(type=genai.protos.Type.STRING),
                        "to_cur": genai.protos.Schema(type=genai.protos.Type.STRING),
                    },
                    required=["from_cur", "to_cur"],
                ),
            ),
        ])
    ]

    agent = ParallelToolAgent(tools=TOOLS)
    agent.register_tool("get_stock", get_stock)
    agent.register_tool("get_news", get_news)
    agent.register_tool("get_fx", get_fx)

    result = await agent.run(
```

## 4. TypeScript — Next.js API Route

```typescript
// app/api/gemini-agent/route.ts
import { GoogleGenerativeAI, FunctionCallingMode } from '@google/generative-ai';
import { NextRequest, NextResponse } from 'next/server';

const genai = new GoogleGenerativeAI(process.env.GOOGLE_API_KEY!);

const TOOLS = [
  {
    functionDeclarations: [
      {
        name: 'search_products',
        description: 'Search product catalog',
        parameters: {
          type: 'object' as const,
          properties: {
            query: { type: 'string', description: 'Search query' },
            maxResults: { type: 'number', description: 'Max results (default 5)' },
          },
          required: ['query'],
        },
      },
      {
        name: 'get_product_details',
        description: 'Get product details by ID',
        parameters: {
          type: 'object' as const,
          properties: {
            productId: { type: 'string', description: 'Product ID' },
          },
          required: ['productId'],
        },
      },
    ],
  },
];

async function searchProducts(query: string, maxResults = 5) {
  return {
    results: Array.from({ length: maxResults }, (_, i) => ({
      id: `prod_${i + 1}`,
      name: `${query} Product ${i + 1}`,
      price: 29.99 + i * 10,
    })),
  };
}

async function getProductDetails(productId: string) {
  return { id: productId, name: `Product ${productId}`, price: 49.99, stock: 12 };
}

const DISPATCH: Record<string, (args: any) => Promise<object>> = {
  search_products: ({ query, maxResults }: any) => searchProducts(query, maxResults),
  get_product_details: ({ productId }: any) => getProductDetails(productId),
};

export async function POST(req: NextRequest) {
  const { message } = await req.json();

  const model = genai.getGenerativeModel({
    model: 'gemini-2.0-flash',
    tools: TOOLS,
    toolConfig: { functionCallingConfig: { mode: FunctionCallingMode.AUTO } },
  });

  const chat = model.startChat();
  let response = await chat.sendMessage(message);

  for (let i = 0; i < 10; i++) {
    const calls = response.response.functionCalls();
    if (!calls?.length) break;

    const results = await Promise.all(
      calls.map(async (fc) => {
        const fn = DISPATCH[fc.name];
        const result = fn ? await fn(fc.args as any) : { error: `Unknown tool: ${fc.name}` };
        return { functionResponse: { name: fc.name, response: result } };
      }),
    );

    response = await chat.sendMessage(results);
  }

  return NextResponse.json({ result: response.response.text() });
}
```
