Company knowledge bases, product documentation, technical specifications — making these searchable while having Gemini generate grounded answers (not hallucinations) is a high-value problem with a clear solution: Vertex AI Search combined with Gemini API's Grounding feature.

Vertex AI Search is Google's managed enterprise search service. It handles vector indexing, scaling, and access control — while Gemini uses those search results as grounded context for generating natural language answers. The result is enterprise-grade RAG without managing a pgvector cluster or custom Chroma server.

## Vertex AI Search Architecture

Three main data store types:

**Unstructured** — PDFs, Word documents, HTML, text files. Semantic search over document content. Best for internal manuals and technical documentation.

**Structured** — JSON/BigQuery data for product catalogs, inventory, and tabular content.

**Website** — Crawl and index public-facing documentation sites by URL pattern.

## Creating a Data Store and Importing Documents

### Create an Unstructured Data Store

```python
from google.cloud import discoveryengine_v1 as discoveryengine

PROJECT_ID = "YOUR_PROJECT_ID"
LOCATION = "global"
DATA_STORE_ID = "company-docs-store"

def create_datastore():
    client = discoveryengine.DataStoreServiceClient()

    parent = f"projects/{PROJECT_ID}/locations/{LOCATION}/collections/default_collection"

    data_store = discoveryengine.DataStore(
        display_name="Company Documentation",
        industry_vertical=discoveryengine.IndustryVertical.GENERIC,
        content_config=discoveryengine.DataStore.ContentConfig.CONTENT_REQUIRED,
        solution_types=[discoveryengine.SolutionType.SOLUTION_TYPE_SEARCH],
    )

    operation = client.create_data_store(
        parent=parent,
        data_store=data_store,
        data_store_id=DATA_STORE_ID,
    )

    return operation.result()
```

### Batch Import from GCS

```python
def import_from_gcs(gcs_uri: str):
    """Import all PDFs from a GCS bucket path, e.g. gs://bucket/docs/"""
    client = discoveryengine.DocumentServiceClient()

    parent = (
        f"projects/{PROJECT_ID}/locations/{LOCATION}"
        f"/collections/default_collection"
        f"/dataStores/{DATA_STORE_ID}/branches/default_branch"
    )

    operation = client.import_documents(
        request=discoveryengine.ImportDocumentsRequest(
            parent=parent,
            gcs_source=discoveryengine.GcsSource(
                input_uris=[gcs_uri],
                data_schema="document",
            ),
            reconciliation_mode=discoveryengine.ImportDocumentsRequest.ReconciliationMode.INCREMENTAL,
        )
    )

    return operation.result(timeout=600)
```

## Semantic Search Implementation

```python
SEARCH_ENGINE_ID = "company-search-app"

def semantic_search(
    query: str,
    page_size: int = 5,
    filter_expr: str | None = None,
) -> tuple[list[dict], str]:
    client = discoveryengine.SearchServiceClient()

    serving_config = (
        f"projects/{PROJECT_ID}/locations/{LOCATION}"
        f"/collections/default_collection"
        f"/engines/{SEARCH_ENGINE_ID}"
        f"/servingConfigs/default_config"
    )

    request = discoveryengine.SearchRequest(
        serving_config=serving_config,
        query=query,
        page_size=page_size,
        filter=filter_expr,
        content_search_spec=discoveryengine.SearchRequest.ContentSearchSpec(
            snippet_spec=discoveryengine.SearchRequest.ContentSearchSpec.SnippetSpec(
                return_snippet=True,
                max_snippet_count=3,
            ),
            extractive_content_spec=discoveryengine.SearchRequest.ContentSearchSpec.ExtractiveContentSpec(
                max_extractive_answer_count=3,
                max_extractive_segment_count=5,
            ),
            summary_spec=discoveryengine.SearchRequest.ContentSearchSpec.SummarySpec(
                summary_result_count=3,
                include_citations=True,
                ignore_adversarial_query=True,
            ),
        ),
        query_expansion_spec=discoveryengine.SearchRequest.QueryExpansionSpec(
            condition=discoveryengine.SearchRequest.QueryExpansionSpec.Condition.AUTO,
        ),
    )

    response = client.search(request)

    results = []
    for result in response.results:
        doc = result.document
        chunks = []

        if doc.derived_struct_data:
            for answer in doc.derived_struct_data.get("extractive_answers", []):
                chunks.append({
                    "type": "answer",
                    "content": answer.get("content", ""),
                    "page": answer.get("pageNumber"),
                })

        results.append({
            "id": doc.id,
            "title": doc.derived_struct_data.get("title", "") if doc.derived_struct_data else "",
            "link": doc.derived_struct_data.get("link", "") if doc.derived_struct_data else "",
            "chunks": chunks,
            "relevance_score": result.relevance_score,
        })

    summary = getattr(response, "summary", None)
    summary_text = summary.summary_text if summary else ""

    return results, summary_text
```

## Gemini Grounding with Vertex AI Search

The simplest path to grounded answers — Gemini automatically queries Vertex AI Search and cites its sources.

```python
import google.generativeai as genai

genai.configure(api_key="YOUR_GEMINI_API_KEY")

def generate_grounded_answer(query: str) -> dict:
    model = genai.GenerativeModel("gemini-2.5-flash")

    grounding_tool = genai.types.Tool(
        retrieval=genai.types.Retrieval(
            vertex_ai_search=genai.types.VertexAISearch(
                datastore=(
                    f"projects/{PROJECT_ID}/locations/{LOCATION}"
                    f"/collections/default_collection"
                    f"/dataStores/{DATA_STORE_ID}"
                ),
            )
        )
    )

    response = model.generate_content(
        query,
        tools=[grounding_tool],
        generation_config=genai.types.GenerationConfig(temperature=0.2),
    )

    grounding_sources = []
    if response.candidates[0].grounding_metadata:
        for chunk in response.candidates[0].grounding_metadata.grounding_chunks:
            if chunk.retrieved_context:
                grounding_sources.append({
                    "title": chunk.retrieved_context.title,
                    "uri": chunk.retrieved_context.uri,
                    "text": chunk.retrieved_context.text,
                })

    return {
        "answer": response.text,
        "sources": grounding_sources,
        "is_grounded": len(grounding_sources) > 0,
    }
```

## Manual RAG Pipeline (Fine-Grained Control)

When you need precise control over retrieval, prompt design, and source selection:

```python
def manual_rag_pipeline(
    user_query: str,
    conversation_history: list[dict] | None = None,
) -> dict:
    # Step 1: Retrieve relevant chunks
    results, search_summary = semantic_search(query=user_query, page_size=5)

    # Step 2: Build context from top chunks
    context = "\n\n".join(
        f"[Document {i+1}: {r['title']}]\n{r['chunks'][0]['content']}"
        for i, r in enumerate(results)
        if r["chunks"]
    )

    # Step 3: Generate grounded answer
    model = genai.GenerativeModel(
        "gemini-2.5-flash",
        system_instruction=(
            "You are a company knowledge assistant. "
            "Answer based only on the provided context. "
            "If the information isn't in the context, say so clearly. "
            "Always cite the document source for your answers."
        ),
    )

    messages = list(conversation_history or [])
    messages.append({
        "role": "user",
        "parts": [f"Context:\n{context}\n\nQuestion: {user_query}"],
    })

    response = model.generate_content(messages)

    return {
        "answer": response.text,
        "sources": [{"title": r["title"], "link": r["link"]} for r in results[:3]],
        "search_summary": search_summary,
    }
```

## Access Control for Enterprise Deployments

```python
def search_with_acl(
    query: str,
    user_email: str,
    user_groups: list[str],
) -> list[dict]:
    accessible_categories = get_user_accessible_categories(
        user_email, user_groups
    )

    if not accessible_categories:
        return []

    filter_expr = " OR ".join(
        f'category = "{cat}"' for cat in accessible_categories
    )

    results, _ = semantic_search(query=query, filter_expr=filter_expr)
    return results
```

## Response Caching

```python
import hashlib, time

_cache: dict[str, dict] = {}
CACHE_TTL = 3600  # 1 hour

def cached_rag(query: str, filter_expr: str | None = None) -> dict:
    key = hashlib.md5(f"{query}:{filter_expr}".encode()).hexdigest()

    if key in _cache:
        entry = _cache[key]
        if time.time() - entry["ts"] < CACHE_TTL:
            return entry["data"]

    data = manual_rag_pipeline(query)
    _cache[key] = {"data": data, "ts": time.time()}
    return data
```

## Closing Thoughts

Vertex AI Search removes the operational burden of running a vector database — no index management, no capacity planning, no backup strategy to design. You get Google-scale search infrastructure with built-in security and compliance.

The Grounding approach is the fastest path to a working system: point Gemini at your Vertex AI Search data store and get cited, grounded answers immediately. For production workloads requiring precise source selection or complex multi-turn reasoning, the manual RAG pipeline gives you full control.

Either way, combining Vertex AI Search's retrieval precision with Gemini's language understanding delivers a knowledge assistant that genuinely helps users find answers — rather than confidently generating plausible-sounding fictions.
