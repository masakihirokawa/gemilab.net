## Notion API Basics

### Initialize the Client and Fetch Database Pages

```python
from notion_client import Client

# Initialize the Notion client
notion = Client(auth=NOTION_TOKEN)

def get_database_pages(database_id: str) -> list:
    """Fetch all pages from a Notion database with pagination support."""
    results = []
    has_more = True
    start_cursor = None

    while has_more:
        response = notion.databases.query(
            database_id=database_id,
            start_cursor=start_cursor,
        )
        results.extend(response["results"])
        has_more = response["has_more"]
        start_cursor = response.get("next_cursor")

    return results

# Example usage
pages = get_database_pages(NOTION_DATABASE_ID)
print(f"Total pages fetched: {len(pages)}")
# Expected output: Total pages fetched: 42
```

### Create a New Page

```python
def create_notion_page(
    database_id: str,
    title: str,
    content: str,
    tags: list[str] = None
) -> dict:
    """Add a new page to a Notion database."""
    properties = {
        "Name": {
            "title": [{"text": {"content": title}}]
        }
    }

    # Add tags if provided
    if tags:
        properties["Tags"] = {
            "multi_select": [{"name": tag} for tag in tags]
        }

    # Page body blocks
    children = [
        {
            "object": "block",
            "type": "paragraph",
            "paragraph": {
                "rich_text": [{"type": "text", "text": {"content": content}}]
            }
        }
    ]

    response = notion.pages.create(
        parent={"database_id": database_id},
        properties=properties,
        children=children
    )
    return response

# Example usage
page = create_notion_page(
    database_id=NOTION_DATABASE_ID,
    title="Gemini API Usage Notes",
    content="Notes on building a new workflow with Gemini 3.1 Pro",
    tags=["AI", "Notes"]
)
print(f"Created page ID: {page['id']}")
```

## Practical Gemini × Notion Workflows

### Workflow 1: Auto-Summarize Web Articles and Save to Notion

Pass a URL and Gemini will summarize the content, then save it to your Notion database.

```python
import google.generativeai as genai
from notion_client import Client
import requests
from bs4 import BeautifulSoup

genai.configure(api_key=GEMINI_API_KEY)
model = genai.GenerativeModel("gemini-2.5-flash-preview-04-17")
notion = Client(auth=NOTION_TOKEN)

def fetch_article_text(url: str) -> str:
    """Scrape and extract the main text content from a URL."""
    headers = {"User-Agent": "Mozilla/5.0"}
    response = requests.get(url, headers=headers, timeout=10)
    soup = BeautifulSoup(response.text, "html.parser")
    article = soup.find("article") or soup.find("main") or soup.body
    return article.get_text(separator="\n", strip=True)[:5000]

def summarize_and_save(url: str, category: str = "General") -> dict:
    """
    Fetch article → summarize with Gemini → save to Notion.
    Returns the Notion page ID and the generated summary.
    """
    # 1. Fetch article text
    article_text = fetch_article_text(url)

    # 2. Generate summary with Gemini
    prompt = f"""Please summarize the following article in English in 150 words or less.
Structure your response with 3–5 key bullet points, followed by a one-sentence takeaway.

[Article Content]
{article_text}
"""
    response = model.generate_content(prompt)
    summary = response.text

    # 3. Save to Notion
    page = create_notion_page(
        database_id=NOTION_DATABASE_ID,
        title=f"[Summary] {url[:50]}...",
        content=f"Source: {url}\n\n{summary}",
        tags=[category, "AI-Summary"]
    )

    print(f"✅ Saved to Notion: {page['id']}")
    return {"page_id": page["id"], "summary": summary}

# Example usage
result = summarize_and_save(
    url="https://ai.google.dev/gemini-api/docs",
    category="AI Tech"
)
# Expected output: ✅ Saved to Notion: abc123...
```

### Workflow 2: AI-Powered Task Prioritization

Gemini reads your task database and automatically assigns priority levels.

```python
import json

def analyze_task_priority(task_title: str, task_description: str) -> dict:
    """
    Use Gemini to analyze a task and return priority, category,
    estimated hours, and reasoning in JSON format.
    """
    prompt = f"""Analyze the following task and respond in JSON format.

Task: {task_title}
Description: {task_description}

Response format (strictly follow this JSON):
{{
  "priority": "high" | "medium" | "low",
  "category": "Development" | "Marketing" | "Operations" | "Research" | "Other",
  "estimated_hours": number (e.g., 2.5),
  "reason": "One-sentence explanation of the priority decision"
}}
"""

    response = model.generate_content(
        prompt,
        generation_config={"response_mime_type": "application/json"}
    )

    return json.loads(response.text)

def auto_prioritize_tasks(database_id: str) -> None:
    """
    Fetch all tasks from a Notion database and auto-assign priorities
    using Gemini API for tasks that don't have one yet.
    """
    pages = get_database_pages(database_id)
    prioritized = 0

    for page in pages:
        title_blocks = page["properties"].get("Name", {}).get("title", [])
        title = title_blocks[0]["text"]["content"] if title_blocks else "Untitled"

        desc_blocks = page["properties"].get("Description", {}).get("rich_text", [])
        description = desc_blocks[0]["text"]["content"] if desc_blocks else ""

        # Skip tasks that already have a priority
        if page["properties"].get("Priority", {}).get("select"):
            continue

        analysis = analyze_task_priority(title, description)

        notion.pages.update(
            page_id=page["id"],
            properties={
                "Priority": {
                    "select": {"name": analysis["priority"]}
                },
                "Category": {
                    "select": {"name": analysis["category"]}
                }
            }
        )
        prioritized += 1
        print(f"✅ {title}: {analysis['priority']} — {analysis['reason']}")

    print(f"\n🎉 Auto-prioritized {prioritized} tasks")

# Example output:
# ✅ Redesign landing page: high — Directly impacts user conversion rates
# ✅ Competitor analysis report: medium — Strategic but not time-critical
# 🎉 Auto-prioritized 12 tasks
```

### Workflow 3: Automated Weekly Progress Reports

Pull last week's completed tasks from Notion, analyze them with Gemini, and save a report.

```python
from datetime import datetime, timedelta

def generate_weekly_report(database_id: str) -> str:
    """
    Fetch completed tasks from the past week and generate a
    structured progress report using Gemini.
    """
    one_week_ago = (datetime.now() - timedelta(days=7)).isoformat()

    response = notion.databases.query(
        database_id=database_id,
        filter={
            "and": [
                {
                    "property": "Status",
                    "select": {"equals": "Done"}
                },
                {
                    "property": "Completed At",
                    "date": {"after": one_week_ago}
                }
            ]
        }
    )

    completed_tasks = []
    for page in response["results"]:
        title_blocks = page["properties"].get("Name", {}).get("title", [])
        title = title_blocks[0]["text"]["content"] if title_blocks else "Untitled"
        completed_tasks.append(title)

    if not completed_tasks:
        return "No tasks were completed this week."

    task_list = "\n".join([f"- {t}" for t in completed_tasks])
    prompt = f"""You are a project manager writing a weekly review.
Based on the completed tasks below, write a concise weekly report with three sections:
1. This Week's Highlights
2. Challenges and Insights
3. Recommendations for Next Week

[Completed Tasks]
{task_list}
"""
    result = model.generate_content(prompt)
    report_text = result.text

    today = datetime.now().strftime("%Y-%m-%d")
    create_notion_page(
        database_id=database_id,
        title=f"Weekly Report {today}",
        content=report_text,
        tags=["Report", "Weekly"]
    )

    return report_text

report = generate_weekly_report(NOTION_DATABASE_ID)
print(report)
```

## Common Errors and Fixes

### Error 1: Notion API 400 Bad Request

```python
# Wrong: The property name doesn't match what's in your database
properties = {
    "Title": {...}  # Your database might use "Name" instead
}

# Fix: Check your actual property names first
db_info = notion.databases.retrieve(database_id=NOTION_DATABASE_ID)

# Output: ['Name', 'Tags', 'Status', 'Priority']
```

### Error 2: Gemini API Rate Limiting (429)

```python
import time

def generate_with_retry(prompt: str, max_retries: int = 3) -> str:
    """Retry logic with exponential backoff for rate limit errors."""
    for attempt in range(max_retries):
        try:
            response = model.generate_content(prompt)
            return response.text
        except Exception as e:
            if "429" in str(e) and attempt < max_retries - 1:
                wait_time = 2 ** attempt  # 1s, 2s, 4s
                print(f"⏳ Rate limited. Retrying in {wait_time}s ({attempt + 1}/{max_retries})")
                time.sleep(wait_time)
            else:
                raise
    return ""
```

### Error 3: Notion Block Limit

Notion's API limits each request to 100 blocks. For long content, split it into chunks.

```python
def create_long_content_page(database_id: str, title: str, content: str) -> dict:
    """Split long content into multiple Notion blocks."""
    chunks = [content[i:i+2000] for i in range(0, len(content), 2000)]

    children = [
        {
            "object": "block",
            "type": "paragraph",
            "paragraph": {
                "rich_text": [{"type": "text", "text": {"content": chunk}}]
            }
        }
        for chunk in chunks
    ]

    return notion.pages.create(
        parent={"database_id": database_id},
        properties={"Name": {"title": [{"text": {"content": title}}]}},
        children=children[:100]  # Maximum 100 blocks per request
    )
```

## FAQ

**Q1. Does this work with Notion's free plan?**

Yes! Notion's free plan supports API integrations. The Gemini API also has a generous free tier (60 requests per minute on Flash models), so you can get started at zero cost for personal or learning projects.

**Q2. How do I read the actual body content (blocks) of a Notion page?**

Use `notion.blocks.children.list(block_id=page_id)` to retrieve all blocks within a page. Filter by `type` (e.g., `paragraph`, `heading_2`) and extract text from the `rich_text` field. For deeply nested blocks, you'll need to fetch children recursively.

**Q3. Can I trigger Gemini processing in real time when a Notion page is updated?**

Notion doesn't currently offer webhooks (as of April 2026). The recommended approach is scheduled polling — using GitHub Actions or Cloud Scheduler to check for changes every 5–15 minutes.
